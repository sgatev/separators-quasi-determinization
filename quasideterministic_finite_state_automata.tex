\section{Quasi-deterministic finite-state devices}

\subsection{Quasi-deterministic finite-state automata}

\begin{theorem}
	Let $\mathcal{A} = \langle \Sigma, Q, I, F, \Delta \rangle$ be a
	finite-state automaton. Let the partition $Q = S \cup Q_1 \cup Q_2$
	be such that $\forall q \in Q_1 \forall \sigma \in \Sigma
	(\Delta^*(q, \sigma) \cap Q_2 = \varnothing)$ and $\forall
	q \in Q_2 \forall \sigma \in \Sigma (\Delta^*(q, \sigma) \cap Q_1 =
	\varnothing)$. We can construct a deterministic finite-state automaton
	$\mathcal{A'}$ such that $\mathcal{L} (\mathcal{A'} ) = \mathcal{L} (
	\mathcal{A} )$.
\end{theorem}

\begin{proof}
	Let $P_1 = Q_1 \cup S$, $P_2 = Q_2 \cup S$. Consider the automaton
	$\mathcal{A'} = \langle \Sigma, Q', i, F', \delta \rangle$ where

	\begin{enumerate}
		\item $Q' = 2^{P_1} \times 2^{P_2}$;
		\item $i = \langle P_1 \cap I, P_2 \cap I \rangle$;
		\item $F' = \{ \langle q', q'' \rangle \in Q' \mid q' \cap F
			\neq \varnothing \lor q'' \cap F \neq \varnothing \}$;
		\item $\forall \langle q', q'' \rangle \in Q'\; \forall \sigma
			\in \Sigma(\delta(\langle q', q'' \rangle, \sigma) =\\
				\langle (\Delta^*(q', \sigma) \cap P_1) \cup (\Delta^*(q'', \sigma) \cap S),
			(\Delta^*(q'', \sigma) \cap P_2) \cup (\Delta^*(q', \sigma) \cap S) \rangle)$.
	\end{enumerate}

	\begin{case}We show that every successful path in $\mathcal{A}$ is also a successful
		path in $\mathcal{A'}$ by induction on the states visited in the path.
		Let $\pi = \langle q_1, \sigma_1, q_2 \rangle \ldots
		\langle q_{n-1}, \sigma_{n-1}, q_n \rangle$ be a successful path
		in $\mathcal{A}$. Then
		\begin{enumerate}
			\item $q_1 \in P_1 \cap I \lor q_1 \in P_2 \cap I \implies q_1 \in q_1' \lor q_1 \in q_1''$ where $\langle q_1', q_1'' \rangle = i$.
			\item Let $\delta^*(i, \sigma_1 \dots \sigma_{i-1}) = \langle q_i', q_i'' \rangle$ and $q_i \in q_i' \lor q_i \in q_i''$.
				Then $\delta(\langle q_i', q_i'' \rangle, \sigma_i) = \langle q_{i+1}', q_{i+1}'' \rangle$ where\\\\
				$\begin{aligned}
					q_{i+1}' &= (\Delta^*(q_i', \sigma_i) \cap P_1) \cup (\Delta^*(q_i'', \sigma_i) \cap S)\\
					&= \Big[ \big[ \bigcup_{q \in q_i'} \Delta(q, \sigma_i) \cup \Delta_\epsilon(q) \big] \cap P_1 \Big] \cup \Big[ \big[ \bigcup_{q \in q_i''} \Delta(q, \sigma_i) \cup \Delta_\epsilon(q) \big]  \cap S \Big]\\
					&= \Big[ \big[ \bigcup_{q \in q_i'} \Delta(q, \sigma_i) \cup \Delta_\epsilon(q) \big] \cap P_1 \Big] \cup \Big[ \big[ \bigcup_{q \in q_i''} \Delta(q, \sigma_i) \cup \Delta_\epsilon(q) \big]  \cap P_1 \Big]\\
					&= \big[ \bigcup_{q \in q_i' \cup q_i''} \Delta(q, \sigma_i) \cup \Delta_\epsilon(q) \big] \cap P_1
				\end{aligned}$\\
				$\begin{aligned}
					q_{i+1}'' &= (\Delta^*(q_i'', \sigma_i) \cap P_2) \cup (\Delta^*(q_i', \sigma_i) \cap S) \rangle)\\
					&= \Big[ \big[ \bigcup_{q \in q_i''} \Delta(q, \sigma_i) \cup \Delta_\epsilon(q) \big] \cap P_2 \Big] \cup \Big[ \big[ \bigcup_{q \in q_i'} \Delta(q, \sigma_i) \cup \Delta_\epsilon(q) \big] \cap S \Big]\\
					&= \Big[ \big[ \bigcup_{q \in q_i''} \Delta(q, \sigma_i) \cup \Delta_\epsilon(q) \big] \cap P_2 \Big] \cup \Big[ \big[ \bigcup_{q \in q_i'} \Delta(q, \sigma_i) \cup \Delta_\epsilon(q) \big] \cap P_2 \Big]\\
					&= \big[ \bigcup_{q \in q_i' \cup q_i''} \Delta(q, \sigma_i) \cup \Delta_\epsilon(q) \big] \cap P_2
				\end{aligned}$\\
				$\begin{aligned}
					q_{i+1} \in \Delta(q_i, \sigma_i) \cup \Delta_\epsilon(q_i)
					&\implies q_{i+1} \in \bigcup_{q \in q_i' \cup q_i''} \Delta(q, \sigma_i) \cup \Delta_\epsilon(q)\\
					&\implies \begin{aligned}[t] &q_{i+1} \in \big[ \bigcup_{q \in q_i' \cup q_i''} \Delta(q, \sigma_i) \cup \Delta_\epsilon(q) \big] \cap P_1 \lor\\
						&q_{i+1} \in \big[ \bigcup_{q \in q_i' \cup q_i''} \Delta(q, \sigma_i) \cup \Delta_\epsilon(q) \big] \cap P_2\\
					\end{aligned}\\
					&\implies  q_{i+1} \in q_{i+1}' \lor q_{i+1} \in q_{i+1}''
				\end{aligned}$
			\item $\begin{aligned}[t] q_n \in F \land \exists \langle q_n', q_n'' \rangle &\in Q'(q_n \in q_n' \lor q_n \in q_n'')\\
					&\implies q_n' \cap F \neq \varnothing \lor q_n'' \cap F \neq \varnothing\\
					&\implies \langle q_n', q_n'' \rangle \in F'
				\end{aligned}$
		\end{enumerate}
		\bigbreak
		\noindent
		$\implies \pi' = \langle \langle q_1', q_1'' \rangle, \sigma_1, \langle q_2', q_2'' \rangle \rangle \dots \langle \langle q_{n-1}', q_{n-1}'' \rangle, \sigma_{n-1}, \langle q_n', q_n'' \rangle \rangle$ is a successful path in $\mathcal{A'}$.
	\end{case}

	\begin{case}We show that every successful path in $\mathcal{A'}$ is also a successful
		path in $\mathcal{A}$ by induction on the states visited in the path.
		Let $\pi' = \langle \langle q_1', q_1'' \rangle, \sigma_1', \langle q_2', q_2'' \rangle \rangle \ldots\\
		\langle \langle q_{n-1}', q_{n-1}'' \rangle, \sigma_{n-1}', \langle q_n', q_n'' \rangle \rangle$ be a successful path in $\mathcal{A'}$.
		\begin{enumerate}
			\item $\begin{aligned}[t] \langle q_n', q_n'' \rangle \in F' &\implies q_n' \cap F \neq \varnothing\;\lor\;q_n'' \cap F \neq \varnothing\\
					&\implies (q_n' \cup q_n'') \cap F \neq \varnothing\\
					&\implies \exists\;q_n \in q_n' \cup q_n''(q_n \in F)\\
				\end{aligned}$
			\item $\begin{aligned}[t] \text{Let } q_{i+1} &\in q_{i+1}' \cup\;q_{i+1}''\\
					&= \beta_1(\gamma_1(q_i', \sigma_i'), \alpha_2(\gamma_2(q_i'', \sigma_i'))) \cup \beta_2(\gamma_2(q_i'', \sigma_i'), \alpha_1(\gamma_1(q_i', \sigma_i')))\\
					&= \gamma_1(q_i', \sigma_i') \cup \alpha_2(\gamma_2(q_i'', \sigma_i')) \cup \gamma_2(q_i'', \sigma_i') \cup \alpha_1(\gamma_1(q_i', \sigma_i'))\\
					&= \begin{aligned}[t] &\big[ \bigcup_{q \in q_i'} \Delta(q, \sigma_i') \cup \Delta_\epsilon(q) \big] \cap P_1\;\cup\\
						&\big[ \bigcup_{q \in q_i''} \Delta(q, \sigma_i') \cup \Delta_\epsilon(q) \big] \cap P_2 \cap S\;\cup\\
						&\big[ \bigcup_{q \in q_i''} \Delta(q, \sigma_i') \cup \Delta_\epsilon(q) \big] \cap P_2\;\cup\\
					&\big[ \bigcup_{q \in q_i'} \Delta(q, \sigma_i') \cup \Delta_\epsilon(q) \big] \cap P_1 \cap S \end{aligned}\\
					&= \begin{aligned}[t] &\big[ \bigcup_{q \in q_i'} \Delta(q, \sigma_i') \cup \Delta_\epsilon(q) \big] \cap P_1\;\cup\\
						&\big[ \bigcup_{q \in q_i''} \Delta(q, \sigma_i') \cup \Delta_\epsilon(q) \big] \cap S\;\cup\\
						&\big[ \bigcup_{q \in q_i''} \Delta(q, \sigma_i') \cup \Delta_\epsilon(q) \big] \cap P_2\;\cup\\
					&\big[ \bigcup_{q \in q_i'} \Delta(q, \sigma_i') \cup \Delta_\epsilon(q) \big] \cap S  \end{aligned}\\
					&= \begin{aligned}[t] &\big[ \bigcup_{q \in q_i'} \Delta(q, \sigma_i') \cup \Delta_\epsilon(q) \big] \cap P_1\;\cup\\
					&\big[ \bigcup_{q \in q_i''} \Delta(q, \sigma_i') \cup \Delta_\epsilon(q) \big] \cap P_2  \end{aligned}\\
					&= \big[ \bigcup_{q \in q_i'} \Delta(q, \sigma_i') \cup \Delta_\epsilon(q) \big] \cup \big[ \bigcup_{q \in q_i''} \Delta(q, \sigma_i') \cup \Delta_\epsilon(q) \big]
				\end{aligned}$
				\bigbreak
				$\implies \begin{aligned}[t] &\exists\;q_i \in q_i' : q_{i+1} \in \Delta(q_i, \sigma_i') \cup \Delta_\epsilon(q_i)\;\lor\\
				&\exists\;q_i \in q_i'' : q_{i+1} \in \Delta(q_i, \sigma_i') \cup \Delta_\epsilon(q_i) \end{aligned}$
				\bigbreak
				$\implies \begin{aligned}[t] &\exists\;q_i \in q_i' : q_{i+1} \in \Delta(q_i, \sigma_i)\;\lor\\
				&\exists\;q_i \in q_i'' : q_{i+1} \in \Delta(q_i, \sigma_i) \text{, where $\sigma_i \in \Sigma \cup \{ \epsilon \}$} \end{aligned}$
				\bigbreak
			\item $\begin{aligned}[t] q_1' &= (Q_1 \cup S) \cap I \implies \forall q \in q_1' : q \in I\\
					q_1'' &= (S \cup Q_2) \cap I \implies \forall q \in q_1'' : q \in I\\
				\end{aligned}$
		\end{enumerate}
		\bigbreak
		\noindent
		$\implies \pi = \langle q_1, \sigma_1, q_2 \rangle \dots \langle q_{n-1}, \sigma_{n-1}, q_n \rangle$ is a successful path in $\mathcal{A}$.
	\end{case}
\end{proof}
