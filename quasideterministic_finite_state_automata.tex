\begin{definition}
	A quasi-deterministic $k$-separated finite-state automaton $\mathcal{A}$
	is a tuple $\langle \Sigma, \vec{Q}, S, i, F, \vec{\kappa},
	\vec{\rho}, \vec{\eta} \rangle$ where $\Sigma$ is an alphabet, $Q_i$ is
	a set of quasi-states for $i \in \interval{1}{k}$, $S$ is such that
	$\forall i \in \interval{1}{k} (S \subseteq Q_i)$ and $\forall i, j \in
	\interval{1}{k}, i \neq j (Q_i \cap Q_j \subseteq S)$, $i \in
	\prod_{i=1}^k{Q_i}$ is an initial state, $F \subseteq \prod_{i=1}^k{Q_i}$
	is a set of final states, $\kappa_i : Q_i \times \Sigma \rightarrow Q_i$ is
	a component quasi-transition function for $i \in \interval{1}{k}$,
	$\rho_i : Q_i \rightarrow S$ is a component reducer function for $i \in
	\interval{1}{k}$ and $\eta_i : Q_i \times S \rightarrow Q_i$ is a component
	extension function for $i \in \interval{1}{k}$.
\end{definition}

\begin{remark}
	Let $\mathcal{A} = \langle \Sigma, \vec{Q}, S, i, F, \vec{\kappa},
	\vec{\rho}, \vec{\eta} \rangle$ be a quasi-deterministic $k$-separat\-ed
	finite-state automaton. We use $q^j$ for $j \in \interval{1}{k}$ to denote
	the $j$-th component of $q = \langle q^1, q^2, \ldots, q^k \rangle \in
	\prod_{i=1}^k{Q_i}$.
\end{remark}

\begin{definition}
	Let $\mathcal{A} = \langle \Sigma, \vec{Q}, S, i, F, \vec{\kappa},
	\vec{\rho}, \vec{\eta} \rangle$ be a quasi-deterministic $k$-separated
	finite-state automaton. The component tansition function $\delta_i :
	\prod_{j=1}^k{Q_j}$\\
	$\times \Sigma \rightarrow Q_i$ is defined inductively such that
	$\delta_i^0(q, \sigma) = \kappa_i(q^i, \sigma)$ for $q \in
	\prod_{j=1}^k{Q_j}$ and $\sigma \in \Sigma$, and $\delta_i^j(q, \sigma) =
	\eta_i(\delta_i^{j-1}(q, \sigma), \rho_j(\kappa_j(q^j, \sigma)))$ for
	$j \in \interval{1}{k}$, $q \in \prod_{j=1}^k{Q_j}$ and $\sigma \in
	\Sigma$.
\end{definition}

\begin{definition}
	Let $\mathcal{A} = \langle \Sigma, \vec{Q}, S, i, F, \vec{\kappa},
	\vec{\rho}, \vec{\eta} \rangle$
	be a quasi-deterministic $k$-separated finite-state automaton. The tansition
	function $\delta : \prod_{i=1}^k{Q_i} \times \Sigma \rightarrow
	\prod_{i=1}^k{Q_i}$ is defined as $\delta(q, \sigma) = \langle
	\delta_1(q, \sigma), \delta_2(q, \sigma), \ldots, \delta_k(q, \sigma)
	\rangle$ for $q \in \prod_{i=1}^k{Q_i}$ and $\sigma \in \Sigma$.
\end{definition}

\begin{definition}
	A path $\pi$ in a quasi-deterministic $k$-separated finite-state
	automaton $\mathcal{A} = \langle \Sigma, \vec{Q}, S, i, F, \vec{\kappa},
	\vec{\rho}, \vec{\eta} \rangle$ is a non-empty sequence
	$\langle q_1, \sigma_1, q_2 \rangle, \langle q_2, \sigma_2, q_3 \rangle,$\\
	$\dots, \langle q_{n-1}, \sigma_{n-1},
	q_n \rangle$ where $\forall i \in \interval{1}{n}(q_i \in
	\prod_{j=1}^k{Q_j})$, $\forall i \in
	\interval{1}{n-1} (\sigma_i \in \Sigma)$ and $\forall i \in
	\interval{1}{n-1} (q_{i+1} =
	\delta(q_i, \sigma_i))$. The path $\pi$ is successful if $q_1 = i$
	and $q_n \in F$. The label of the path $\pi$ is the word
	$\sigma_1\sigma_2\ldots\sigma_{n-1}$ of length $n-1$ over $\Sigma$.
\end{definition}

\begin{definition}
	Let $\mathcal{A} = \langle \Sigma, \vec{Q}, S, i, F, \vec{\kappa},
	\vec{\rho}, \vec{\eta} \rangle$
	be a quasi-deterministic $k$-separated finite-state automaton. The
	generalised function of transitions $\delta^* : \prod_{j=1}^k{Q_j}
	\times \Sigma^* \rightarrow \prod_{j=1}^k{Q_j}$ is defined as the least
	function such that $\delta^*(q, \epsilon) = q$ for $q \in
	\prod_{j=1}^k{Q_j}$ and $\delta^*(q, \sigma w) = \delta^*(\langle
	\delta_1(q^1, \sigma), \ldots, \delta_k(q^k, \sigma) \rangle, w)$
	for $q \in \prod_{j=1}^k{Q_j}$, $\sigma \in \Sigma$ and $w \in \Sigma^*$.
\end{definition}

\begin{definition}
	The language recognised by a quasi-deterministic $k$-separated finite-state
	automaton $\mathcal{A} = \langle \Sigma, \vec{Q}, S, i, F, \vec{\kappa},
	\vec{\rho}, \vec{\eta} \rangle$ is the set of all labels of successful
	paths $\mathcal{L}(\mathcal{A}) = \{ w \in \Sigma^* \mid \delta^*(i, w) \in
	F \}$.
\end{definition}

\begin{theorem}
	Let $\mathcal{A} = \langle \Sigma, Q, I, F, \Delta \rangle$ be a
	$k$-separated finite-state automaton where $Q = \bigcup_{i=1}^k{Q_i}
	\cup S$. We can construct a quasi-deterministic $k$-separated
	finite-state automaton $\mathcal{A'}$ such that $\mathcal{L}
	(\mathcal{A'} ) = \mathcal{L} ( \mathcal{A} )$.
\end{theorem}

\begin{proof}
	Consider the quasi-deterministic $k$-separated finite-state automaton
	$\mathcal{A'} = \langle \Sigma, \vec{Q'}, 2^S, i, F', \vec{\kappa},
	\vec{\rho}, \vec{\eta} \rangle$ where

	\begin{enumerate}
		\item $P_i = Q_i \cup S$ for $i \in \interval{1}{k}$;
		\item $Q_i' = 2^{P_i}$ for $i \in \interval{1}{k}$;
		\item $i = \langle P_1 \cap I, P_2 \cap I, \dots, P_k \cap I
			\rangle$;
		\item $F' = \{ q \in \prod_{i=1}^k{Q_i'} \mid
			\exists i \in \interval{1}{k} (q^i \cap F \neq
			\varnothing) \}$;
		\item $\kappa_i(q, \sigma) = \hat{\Delta}(q, \{ \sigma \})
			\cap P_i$ where $q \in Q_i'$ and $\sigma
			\in \Sigma$ for $i \in \interval{1}{k}$;
		\item $\rho_i(q) = q \cap S$ where $q \in Q_i'$ for $i \in
			\interval{1}{k}$;
		\item $\eta_i(q, s) = q \cup s$ where $q \in Q_i'$ and $s \in 2^S$
			for $i \in \interval{1}{k}$.
	\end{enumerate}
	We show that every successful path in $\mathcal{A}$ is also a successful
	path in $\mathcal{A'}$ by induction on the states present in the path.
	Let $\pi = \langle q_1, \sigma_1, q_2 \rangle, \ldots, \langle q_{n-1},
	\sigma_{n-1}, q_n \rangle$ be a successful path in $\mathcal{A}$. Then
	\begin{enumerate}
		\item Let $q_1' = i$. Then $q_1 \in I \implies q_1 \in
			\bigcup_{i=1}^k{P_i \cap I} \implies q_1 \in
			\bigcup_{i=1}^k{q_1^i}$;
		\item Let $q_i' = \delta^*(i, \sigma_1 \ldots \sigma_{i-1})$ and
			$q_i \in \bigcup_{j=1}^k{q_i^j}$. We have
			$q_{i+1}' = \langle \delta_1(q_i, \sigma_i), \ldots,$\\
			$\delta_k(q_i, \sigma_i) \rangle$ where $q_{i+1}^j =
			(\hat{\Delta}^*(
			q_i^j, \{ \sigma_i \}) \cap P_j) \cup (\bigcup_{r = 1, r
			\neq j}^k{\hat{\Delta}^*(q_i^r, \{ \sigma_i \}) \cap S})
			= (\hat{\Delta}^*(q_i^j, \{ \sigma_i \}) \cap P_j) \cup
			(\bigcup_{r = 1, r \neq i}^k{\hat{\Delta}^*(q_i^r, \{
			\sigma_i \}) \cap P_j}) = \bigcup_{r = 1}^k{\hat{\Delta}^*(
			q_i^r, \{ \sigma_i \}) \cap P_j}$ for $j \in
			\interval{1}{k}$. Then $q_{i+1} \in \Delta^*(q_i,
			\sigma_i) \implies q_{i+1} \in \bigcup_{r=1}^k{
			\hat{\Delta}^*(q_i^r, \{ \sigma_i \})}\\
			\implies q_{i+1} \in \bigcup_{j=1}^k{\bigcup_{r=1}^k{
			\hat{\Delta}^*(q_i^r, \{ \sigma_i \}) \cap P_j}}
			\implies q_{i+1} \in \bigcup_{j=1}^k{q_{i+1}^j}$;
		\item $q_n \in F \land \exists q_n' \in \prod_{i=1}^k{2^{P_i}}
			(q_n \in \bigcup_{j=1}^k{q_n^j})
			\implies \exists j \in \interval{1}{k}(q_n \in q_n^j)$\\
			$\implies q_n' \in F'$.
	\end{enumerate}
	\bigbreak
	\noindent
	$\implies \pi' = \langle q_1', \sigma_1, q_2' \rangle, \dots, \langle q_{n-1}', \sigma_{n-1}, q_n' \rangle$ is a successful path in $\mathcal{A'}$.
	\bigbreak
	\noindent
	Similarly, we show that every successful path in $\mathcal{A'}$ is also a
	successful path in $\mathcal{A}$. Let $\pi' = \langle q_1', \sigma_1, q_2'
	\rangle, \ldots, \langle q_{n-1}', \sigma_{n-1}, q_n' \rangle$ be a
	successful path in $\mathcal{A'}$. Then
	\begin{enumerate}
		\item $q_n' \in F' \implies \exists j \in \interval{1}{k} (q_n^j
			\cap F \neq \varnothing) \implies \exists q_n \in q_n^j
			(q_n \in F)$;
		\item Let $q_{i+1}' = \langle q_{i+1}^1, \ldots, q_{i+1}^k \rangle$
			and $\exists j \in \interval{1}{k} \exists q_{i+1} \in Q
			(q_{i+1} \in q_{i+1}^j)$. We have $q_{i+1}^j = \delta_j(q_i',
			\sigma_i) = (\hat{\Delta}^*(q_i^j, \{ \sigma_i \}) \cap P_j)
			\cup (\bigcup_{r = 1, r \neq j}^k{\hat{\Delta}^*(q_i^r,
			\{ \sigma_i \}) \cap S}) = (\hat{\Delta}^*(q_i^j, \{
			\sigma_i \}) \cap P_j) \cup (\bigcup_{r = 1, r \neq j}^k{
			\hat{\Delta}^*(q_i^r, \{ \sigma_i \}) \cap P_j}) =
			\bigcup_{r = 1}^k{\hat{\Delta}^*( q_i^r, \{ \sigma_i \})
			\cap P_j}$. Then $\exists j \in \interval{1}{k} \exists
			q_{i+1} \in q_{i+1}^j \implies |q_{i+1}^j| > 0 \implies
			|\bigcup_{r = 1}^k{\hat{\Delta}^*( q_i^r, \{ \sigma_i \})
			\cap P_j}|$\\ $> 0 \implies \exists r \in \interval{1}{k}
			\exists q_i \in Q (q_i \in q_i^r)$;
		\item $\exists j \in \interval{1}{k} \exists q_1 \in Q (q_1 \in
			q_1^j) \implies q_1 \in P_j \cap I \implies q_1 \in I$.
	\end{enumerate}
	\bigbreak
	\noindent
	$\implies \pi = \langle q_1, \sigma_1, q_2 \rangle, \ldots, \langle q_{n-1},
	\sigma_{n-1}, q_n \rangle$ is a successful path in $\mathcal{A}$.
\end{proof}
