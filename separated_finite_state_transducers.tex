\section{Separators in finite-state transducers}

In this section, we define the notion of separated finite-state transducers. We then
observe some of the key properties and use them to derive more efficient procedures
for determinization and minimization of such devices.

\subsection{Separated finite-state transducers}

\begin{definition}
	A finite-state transducer $\mathcal{T} = \langle \Sigma, \Omega, Q, S,
	F, \Delta \rangle$ is $k$-separated for $k \in \mathbb{N}$ if and
	only if there exists a separator $X$ in the graph $\mathcal{G}_\mathcal{T}
	= \langle Q, \Sigma \times \Omega^*, \Delta \rangle$ inducing a partition
	$Q_X = \{ Q_1, Q_2, \ldots, Q_k\}$. We write $\mathcal{T} = \langle \Sigma,
	\Omega, \vec{Q}, X,$\\
	$S, F, \Delta \rangle$ to denote the $k$-separated finite-state
	transducer $\mathcal{T}$.
\end{definition}

\begin{proposition}
	Let $\mathcal{T} = \langle \Sigma, \Omega, \vec{Q}, X, S, F, \Delta \rangle$
	be a $k$-separated finite-state transducer. Then $\Delta \subseteq
	\bigcup_{i=1}^k (Q_i \cup X) \times (\Sigma \times \Omega^*) \times (Q_i
	\cup X)$.
\end{proposition}

\begin{proof}
	$\newline$
	$\begin{aligned}
		\Delta &\subseteq Q \times (\Sigma \times \Omega^*) \times Q =
		(\bigcup_{i=1}^k{Q_i} \cup X) \times (\Sigma \times \Omega^*)
		\times (\bigcup_{i=1}^k{Q_i} \cup X)\\
		&= \begin{aligned}[t]
		&\bigcup_{i=1}^k \bigcup_{j=1}^k (Q_i \times (\Sigma \times
		\Omega^*) \times Q_j) \cup\\
		&\bigcup_{i=1}^k (X \times (\Sigma \times
		\Omega^*) \times Q_i) \cup (Q_i \times (\Sigma \times \Omega^*)
		\times X) \cup\\
		&(X \times (\Sigma \times \Omega^*) \times X)
		\end{aligned}\\
		&\subseteq \begin{aligned}[t]
		\bigcup_{i=1}^k &(Q_i \times (\Sigma \times \Omega^*) \times Q_i)
		\cup (X \times (\Sigma \times \Omega^*) \times Q_i) \cup\\
		&(Q_i \times (\Sigma \times \Omega^*) \times X) \cup
		(X \times (\Sigma \times \Omega^*) \times X)
		\end{aligned}\\
		&= \bigcup_{i=1}^k \big[ (Q_i \cup X) \times (\Sigma \times
		\Omega^*) \times Q_i \big] \cup \big[ (Q_i \cup X) \times (\Sigma
		\times \Omega^*) \times X \big]\\
		&= \bigcup_{i=1}^k (Q_i \cup X) \times (\Sigma \times \Omega^*)
		\times (Q_i \cup X)
	\end{aligned}$
	$\newline$
\end{proof}

\begin{proposition}
	Let $\mathcal{T} = \langle \Sigma, \Omega, \vec{Q}, X, S, F, \Delta \rangle$
	be a $k$-separated finite-state transducer. Let $\Delta_i = \Delta \cap
	\big[ (Q_i \cup X) \times (\Sigma \times \Omega^*) \times (Q_i \cup X)
	\big]$ for $i \in \interval{1}{k}$. Then for $i \in \interval{1}{k}$, $q
	\in Q_i \cup X$, $\sigma \in \Sigma$ and $w \in \Omega^*$ we have $\Delta(q,
	\sigma, w) = \Delta_i(q, \sigma, w)$.
\end{proposition}

\begin{proof}
	We have $p \in \Delta(q, \sigma, w) \implies \langle q, \sigma, w, p \rangle
	\in \Delta \cap \big[ \bigcup_{i=1}^k (Q_i \cup X) \times (\Sigma \times
	\Omega^*) \times (Q_i \cup X) \big] \implies \exists i \in
	\interval{1}{k}(\langle q, \sigma, w, p \rangle \in \Delta \cap \big[ (Q_i
	\cup X) \times (\Sigma \times \Omega^*) \times (Q_i \cup X) \big]) \implies
	p \in \Delta_i(q, \sigma, w)$. Similarly, $p \in \Delta_i(q, \sigma, w)$ for
	$i \in \interval{1}{k} \implies p \in \bigcup_{i=1}^k \Delta_i(q, \sigma, w)
	\implies p \in \Delta(q, \sigma, w)$.
\end{proof}

\begin{proposition}
	Let $\mathcal{T} = \langle \Sigma, \Omega, \vec{Q}, X, S, F, \Delta \rangle$
	be a $k$-separated finite-state transducer. Let $\Delta_i = \Delta \cap
	\big[ (Q_i \cup X) \times (\Sigma \times \Omega^*) \times (Q_i \cup X)
	\big]$ for $i \in \interval{1}{k}$. Then for $P \subseteq Q$,  $\sigma \in
	\Sigma$ and $w \in \Omega^*$ we have $\hat{\Delta}(P, \{ \sigma \}, \{ w \})
	= \bigcup_{i=1}^k \hat{\Delta}_i(P_i, \{ \sigma \}, \{ w \})$ where $P_i =
	P \cap (Q_i \cup X)$ for $i \in \interval{1}{k}$.
\end{proposition}

\begin{proof}
	$\newline$
	$\begin{aligned}
		\hat{\Delta}(P, \{ \sigma \}, \{ w \})
		&= \bigcup_{p \in P} \Delta(p, \sigma, w)
		= \bigcup_{i=1}^k \bigcup_{p \in P_i} \Delta(p, \sigma, w)
		= \bigcup_{i=1}^k \bigcup_{p \in P_i} \Delta_i(p, \sigma, w)\\
		&= \bigcup_{i=1}^k \hat{\Delta}_i(P_i, \{ \sigma \}, \{ w \})
	\end{aligned}$
	$\newline$
\end{proof}

\subsection{Quasi-subsequential finite-state transducers}

\begin{definition}
	A quasi-subsequential $k$-separated finite-state transducer $\mathcal{T}$
	is a tuple $\langle \Sigma, \Omega, \vec{Q}, X, s, \vec{f}, \vec{\tau},
	\vec{\rho}, \vec{\eta}, \eta_x, \vec{\mu} \rangle$ where $k \in \mathbb{N}$,
	$\Sigma$ and $\Omega$ are alphabets, $Q_i$ is a set of quasi-states for $i
	\in \interval{1}{k}$, $X$ is a non-empty separator set such that $\forall i
	\in \interval{1}{k} (X \subseteq Q_i)$ and $\forall i, j \in \interval{1}{k},
	i \neq j (Q_i \cap Q_j \subseteq X)$, $s \in \prod_{i=1}^k{Q_i}$ is a start
	state, $f_i : Q_i \rightarrow \Omega^*$ is a final quasi-state output
	function for $i \in \interval{1}{k}$, $\tau_i : Q_i \times \Sigma
	\rightarrow Q_i$ is a component quasi-transition function for $i \in
	\interval{1}{k}$, $\rho_i : Q_i \rightarrow X$ is a component reduction
	function for $i \in \interval{1}{k}$, $\eta_i : Q_i \times X \rightarrow
	Q_i$ is a component extension function for $i \in \interval{1}{k}$, $\eta_x
	: \prod_{i=1}^k{X} \rightarrow X$ is a separator extension function and
	$\mu_i : Q_i \times \Sigma \rightarrow \Omega^*$ is a component quasi-output
	function for $i \in \interval{1}{k}$.
\end{definition}

\begin{remark}
	Let $\mathcal{T} = \langle \Sigma, \Omega, \vec{Q}, X, s, \vec{f},
	\vec{\tau}, \vec{\rho}, \vec{\eta}, \eta_x, \vec{\mu} \rangle$ be a
	quasi-subsequential $k$-separated finite-state transducer. We use $q^j$ for
	$j \in \interval{1}{k}$ to denote the $j$-th component of $q = \langle q^1,
	q^2, \ldots, q^k \rangle \in \prod_{i=1}^k{Q_i}$.
\end{remark}

\begin{definition}
	Let $\mathcal{T} = \langle \Sigma, \Omega, \vec{Q}, X, s, \vec{f},
	\vec{\tau}, \vec{\rho}, \vec{\eta}, \eta_x, \vec{\mu} \rangle$ be a
	quasi-subsequential $k$-separated finite-state transducer. The separator
	tansition function $\delta_x : \prod_{i=1}^k{Q_i}$\\
	$\times \Sigma \rightarrow X$ is defined as $\delta_x(q, \sigma) =
	\eta_x(\rho_1(\tau_1(q^1, \sigma)), \rho_2(\tau_2(q^2, \sigma)), \ldots,
	\rho_k(\tau_k(q^k, \sigma)))$ for $q \in \prod_{i=1}^k{Q_i}$ and $\sigma
	\in \Sigma$.
\end{definition}

\begin{definition}
	Let $\mathcal{T} = \langle \Sigma, \Omega, \vec{Q}, X, s, \vec{f},
	\vec{\tau}, \vec{\rho}, \vec{\eta}, \eta_x, \vec{\mu} \rangle$ be a
	quasi-subsequential $k$-separated finite-state transducer. The component
	tansition functions $\delta_i : \prod_{j=1}^k{Q_j}$\\
	$\times \Sigma \rightarrow Q_i$ are defined as $\delta_i(q, \sigma) =
	\eta_i(\tau_i(q^i, \sigma), \delta_x(q, \sigma))$ for $i \in
	\interval{1}{k}$, $q \in \prod_{j=1}^k{Q_j}$ and $\sigma \in \Sigma$.
\end{definition}

\begin{definition}
	Let $\mathcal{T} = \langle \Sigma, \Omega, \vec{Q}, X, s, \vec{f},
	\vec{\tau}, \vec{\rho}, \vec{\eta}, \eta_x, \vec{\mu} \rangle$ be a
	quasi-subsequential $k$-separated finite-state transducer. The tansition
	function $\delta : \prod_{i=1}^k{Q_i} \times \Sigma \rightarrow
	\prod_{i=1}^k{Q_i}$ is defined as $\delta(q, \sigma) = \langle \delta_1(q,
	\sigma), \delta_2(q, \sigma), \ldots, \delta_k(q, \sigma) \rangle$ for $q
	\in \prod_{i=1}^k{Q_i}$ and $\sigma \in \Sigma$.
\end{definition}

\begin{definition}
	A path $\pi$ in a quasi-subsequential $k$-separated finite-state
	transducer $\mathcal{T} = \langle \Sigma, \Omega, \vec{Q}, X, s, \vec{f},
	\vec{\tau}, \vec{\rho}, \vec{\eta}, \eta_x, \vec{\mu} \rangle$ is a
	non-empty sequence $\langle q_1, \sigma_1, w_1,$\\
	$q_2 \rangle, \langle q_2, \sigma_2, w_2, q_3 \rangle, \ldots,
	\langle q_{n-1}, \sigma_{n-1}, w_{n-1}, q_n \rangle$ where $\forall i \in
	\interval{1}{n}(q_i \in \prod_{j=1}^k{Q_j})$, $\forall i \in
	\interval{1}{n-1} (\sigma_i \in \Sigma)$, $\forall i \in
	\interval{1}{n-1} (w_i \in \Omega^*)$ and $\forall i \in
	\interval{1}{n-1} (q_{i+1} = \delta(q_i, \sigma_i))$. The path $\pi$ is
	successful if and only if $q_1 = s$ and $\exists j \in \interval{1}{k}
	\exists w_n \in \Omega^*(f_j(q_n^j) = w_n)$. The label of the path $\pi$ is
	the word $\sigma_1\sigma_2\ldots\sigma_{n-1}$ of length $n-1$ over $\Sigma$.
	The output of the path $\pi$ is the word $w_1 w_2 \ldots w_{n-1} w_n$ of
	length $\sum_{i=1}^{n}{|w_i|}$ over $\Omega$.
\end{definition}

\begin{definition}
	Let $\mathcal{T} = \langle \Sigma, \Omega, \vec{Q}, X, s, \vec{f},
	\vec{\tau}, \vec{\rho}, \vec{\eta}, \eta_x, \vec{\mu} \rangle$
	be a quasi-subsequential $k$-separated finite-state transducer. The
	generalized function of transitions $\delta^* : \prod_{i=1}^k{Q_i}
	\times \Sigma^* \rightarrow \prod_{i=1}^k{Q_i}$ is defined inductively as
	the least function such that $\delta^*(q, \epsilon) = q$ for $q \in
	\prod_{i=1}^k{Q_i}$ and $\delta^*(q, w\sigma) = \delta(\delta^*(q, w),
	\sigma)$ for $q \in \prod_{i=1}^k{Q_i}$, $w \in \Sigma^*$ and $\sigma \in
	\Sigma$.
\end{definition}

\begin{definition}
	Let $\mathcal{T} = \langle \Sigma, \Omega, \vec{Q}, X, s, \vec{f},
	\vec{\tau}, \vec{\rho}, \vec{\eta}, \eta_x, \vec{\mu} \rangle$ be a
	quasi-subsequential $k$-separated finite-state transducer. The tansition
	output function $\lambda : \prod_{i=1}^k{Q_i} \times \Sigma \rightarrow
	\Omega^*$ is defined as $\lambda(q, \sigma) = \bigwedge_{i=1}^k \mu_i(q^i,
	\sigma)$ for $q \in \prod_{i=1}^k{Q_i}$ and $\sigma \in \Sigma$.
\end{definition}

\begin{definition}
	Let $\mathcal{T} = \langle \Sigma, \Omega, \vec{Q}, X, s, \vec{f},
	\vec{\tau}, \vec{\rho}, \vec{\eta}, \eta_x, \vec{\mu} \rangle$
	be a quasi-subsequential $k$-separated finite-state transducer. The
	generalized function of transition outputs $\lambda^* : \prod_{i=1}^k{Q_i}
	\times \Sigma^* \rightarrow \Omega^*$ is defined inductively as the least
	function such that $\lambda^*(q, \epsilon) = \epsilon$ for $q \in
	\prod_{i=1}^k{Q_i}$ and $\lambda^*(q, w\sigma) = \lambda^*(q, w)\lambda
	(\delta^*(q, w), \sigma)$ for $q \in \prod_{i=1}^k{Q_i}$, $w \in \Sigma^*$
	and $\sigma \in \Sigma$.
\end{definition}

\begin{definition}
	The relation recognized by a quasi-subsequential $k$-separated finite-state
	transducer $\mathcal{T} = \langle \Sigma, \Omega, \vec{Q}, X, s, \vec{f},
	\vec{\tau}, \vec{\rho}, \vec{\eta}, \eta_x, \vec{\mu} \rangle$ is the set
	of all labels and outputs of successful paths $\mathcal{R}(\mathcal{T}) = \{
	\langle u, \lambda^*(s, u) \circ v \rangle \mid \exists q \in
	\prod_{i=1}^k{Q_i} \exists j \in \interval{1}{k}(\delta^*(s, u) = q \land
	f_j(q^j) = v) \}$.
\end{definition}

\begin{theorem}
	Let $\mathcal{T} = \langle \Sigma, \Omega, \vec{Q}, X, S, F, \Delta \rangle$
	be a $k$-separated finite-state transducer. We can construct a
	quasi-subsequential $k$-separated finite-state transducer $\mathcal{T'}$
	such that $\mathcal{R}(\mathcal{T'}) = \mathcal{R}(\mathcal{T})$.
\end{theorem}
