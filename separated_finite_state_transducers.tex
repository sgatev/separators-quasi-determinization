\subsection{Separated finite-state transducers}

\iffalse
\begin{definition}
	A finite-state transducer $\mathcal{T} = \langle \Sigma, \Omega, Q, S, F,
	\Delta \rangle$ is $k$-separated for $k > 1$ if and only if there exists a
	separator $X$ in the graph $\mathcal{G}_\mathcal{T}$ inducing a partition
	$Q_X = \{ Q_1, Q_2, \ldots, Q_k\}$. A $k$-separated finite-state transducer
	$\mathcal{T}$ is denoted by
	\begin{align*}
		\mathcal{T} = \langle \Sigma, \Omega, \vec{Q}, X, S, F, \Delta
		\rangle.
	\end{align*}
\end{definition}

\begin{proposition}
	Let $\mathcal{T} = \langle \Sigma, \Omega, \vec{Q}, X, S, F, \Delta \rangle$
	be a $k$-separated finite-state transducer. Then $\Delta \subseteq
	\bigcup_{i=1}^k (Q_i \cup X) \times (\Sigma_\epsilon \times \Omega^*) \times
	(Q_i \cup X)$.
\end{proposition}

\begin{proof}
	\begin{flalign*}
		\Delta &\subseteq Q \times (\Sigma_\epsilon \times \Omega^*) \times
		Q &\\
		&= (\bigcup_{i=1}^k{Q_i} \cup X) \times (\Sigma_\epsilon \times
		\Omega^*) \times (\bigcup_{i=1}^k{Q_i} \cup X)\\
		&= \begin{aligned}[t]
		&\bigcup_{i=1}^k \bigcup_{j=1}^k (Q_i \times (\Sigma_\epsilon \times
		\Omega^*) \times Q_j) \cup\\
		&\bigcup_{i=1}^k (X \times (\Sigma_\epsilon \times
		\Omega^*) \times Q_i) \cup (Q_i \times (\Sigma_\epsilon \times
		\Omega^*) \times X) \cup\\
		&(X \times (\Sigma_\epsilon \times \Omega^*) \times X).
		\end{aligned}
	\end{flalign*}
	However, $\Delta \cap (Q_i \times (\Sigma_\epsilon \times \Omega^*)
	\times Q_j) = \emptyset$ if $i \neq j$ for $i, j \in \interval{1}{k}$.
	Therefore,
	\begin{flalign*}
		\Delta &\subseteq \begin{aligned}[t]
		&\bigcup_{i=1}^k (Q_i \times (\Sigma_\epsilon \times \Omega^*)
		\times Q_i) \cup\\
		&\bigcup_{i=1}^k (X \times (\Sigma_\epsilon \times \Omega^*) \times
		Q_i) \cup (Q_i \times (\Sigma_\epsilon \times \Omega^*) \times X)
		\cup\\
		&(X \times (\Sigma_\epsilon \times \Omega^*) \times X)
		\end{aligned} &\\
		&= \bigcup_{i=1}^k \big[ (Q_i \cup X) \times (\Sigma_\epsilon \times
		\Omega^*) \times Q_i \big] \cup \big[ (Q_i \cup X) \times
		(\Sigma_\epsilon \times \Omega^*) \times X \big]\\
		&= \bigcup_{i=1}^k (Q_i \cup X) \times (\Sigma_\epsilon \times
		\Omega^*) \times (Q_i \cup X).
	\end{flalign*}
\end{proof}

Based on this result we can modify the construction from theorem
\ref{t_choffrut_mohri} to take advantage of the fact that the state transition
function and the state transition output function of the resulting subsequential
finite-state transducer could be represented as the union of $k$ functions with
smaller domains.
\fi
