\subsection{Separated finite-state transducers}

\begin{definition}
	Let $\mathcal{T} = \langle \Sigma, \Omega, Q, S, F, \Delta \rangle$ be a
	finite-state transducer. $\mathcal{T}$ is $k$-separated if there exists a
	partition $Q = \bigcup_{i=1}^k{Q_i} \cup X$ such that $\forall i, j
	\in \interval{1}{k} (i \neq j \implies \Delta \cap (Q_i \times \Sigma
	\times \Omega^* \times Q_j) = \varnothing)$.
\end{definition}

\begin{definition}
	A quasi-subsequential $k$-separated finite-state transducer $\mathcal{T}$
	is a tuple $\langle \Sigma, \Omega, \vec{Q}, X, s, F, \vec{\kappa},
	\vec{\rho}, \vec{\eta}, \vec{\lambda} \rangle$ where $\Sigma$ is an
	alphabet, $Q_i$ is a set of quasi-states for $i \in \interval{1}{k}$, $X$
	is a non-empty set such that $\forall i \in \interval{1}{k} (X \subseteq
	Q_i)$ and $\forall i, j \in \interval{1}{k}, i \neq j (Q_i \cap Q_j
	\subseteq X)$, $s \in \prod_{i=1}^k{Q_i}$ is a start state, $F \subseteq
	\prod_{i=1}^k{Q_i}$ is a set of final states, $\kappa_i : Q_i \times \Sigma
	\rightarrow Q_i$ is a component quasi-transition function for $i \in
	\interval{1}{k}$, $\rho_i : Q_i \rightarrow X$ is a component reducer
	function for $i \in \interval{1}{k}$, $\eta_i : Q_i \times X \rightarrow
	Q_i$ is a component extension function for $i \in \interval{1}{k}$ and
	$\lambda_i : Q_i \times \Sigma \rightarrow \Omega^*$ is a component output
	function.
\end{definition}

\begin{remark}
	Let $\mathcal{T} = \langle \Sigma, \Omega, \vec{Q}, X, s, F, \vec{\kappa},
	\vec{\rho}, \vec{\eta}, \vec{\lambda} \rangle$ be a quasi-subsequential
	$k$-separated finite-state transducer. We use $q^j$ for $j \in
	\interval{1}{k}$ to denote the $j$-th component of $q = \langle q^1, q^2,
	\ldots, q^k \rangle \in \prod_{i=1}^k{Q_i}$.
\end{remark}

\begin{definition}
	Let $\mathcal{T} = \langle \Sigma, \Omega, \vec{Q}, X, s, F, \vec{\kappa},
	\vec{\rho}, \vec{\eta}, \vec{\lambda} \rangle$ be a quasi-subsequential
	$k$-separated finite-state transducer. The component tansition function
	$\delta_i : \prod_{j=1}^k{Q_j}$\\
	$\times \Sigma \rightarrow Q_i$ is defined as
	$\delta_i^k$ where $\delta_i^j : \prod_{r=1}^k{Q_r} \times \Sigma
	\rightarrow Q_i$ is defined inductively for $j \in \interval{0}{k}$ such
	that $\delta_i^0(q, \sigma) = \kappa_i(q^i, \sigma)$ for $q \in
	\prod_{r=1}^k{Q_r}$ and $\sigma \in \Sigma$, and $\delta_i^j(q, \sigma) =
	\eta_i(\delta_i^{j-1}(q, \sigma), \rho_j(\kappa_j(q^j, \sigma)))$ if $j
	\neq i$, $\delta_i^j(q, \sigma) = \delta_i^{j-1}(q, \sigma)$ if $j = i$ for
	$j \in \interval{1}{k}$, $q \in \prod_{r=1}^k{Q_r}$ and $\sigma \in \Sigma$.
\end{definition}

\begin{definition}
	Let $\mathcal{T} = \langle \Sigma, \Omega, \vec{Q}, X, s, F, \vec{\kappa},
	\vec{\rho}, \vec{\eta}, \vec{\lambda} \rangle$ be a quasi-subsequential
	$k$-separated finite-state transducer. The tansition function $\delta :
	\prod_{i=1}^k{Q_i} \times \Sigma \rightarrow \prod_{i=1}^k{Q_i}$ is defined
	as $\delta(q, \sigma) = \langle \delta_1(q, \sigma), \delta_2(q, \sigma),
	\ldots, \delta_k(q, \sigma) \rangle$ for $q \in \prod_{i=1}^k{Q_i}$ and
	$\sigma \in \Sigma$.
\end{definition}

\begin{definition}
	A path $\pi$ in a quasi-subsequential $k$-separated finite-state
	transducer $\mathcal{T} = \langle \Sigma, \Omega, \vec{Q}, X, s, F,
	\vec{\kappa}, \vec{\rho}, \vec{\eta}, \vec{\lambda} \rangle$ is a non-empty
	sequence $\langle q_1, \sigma_1, w_1, q_2 \rangle,$\\
	$\langle q_2, \sigma_2, w_2, q_3 \rangle, \ldots, \langle q_{n-1},
	\sigma_{n-1}, w_{n-1}, q_n \rangle$ where $\forall i \in
	\interval{1}{n}(q_i \in \prod_{j=1}^k{Q_j})$, $\forall i \in
	\interval{1}{n-1} (\sigma_i \in \Sigma)$, $\forall i \in
	\interval{1}{n-1} (w_i \in \Omega^*)$ and $\forall i \in
	\interval{1}{n-1} (q_{i+1} = \delta(q_i, \sigma_i))$. The path $\pi$ is
	successful if $q_1 = s$ and $q_n \in F$. The label of the path $\pi$ is the
	word $\sigma_1\sigma_2\ldots\sigma_{n-1}$ of length $n-1$ over $\Sigma$.
	The output of the path $\pi$ is the word $w_1 w_2 \ldots w_{n-1}$ over
	$\Omega$.
\end{definition}

\begin{definition}
	Let $\mathcal{T} = \langle \Sigma, \Omega, \vec{Q}, X, s, F,
	\vec{\kappa}, \vec{\rho}, \vec{\eta}, \vec{\lambda} \rangle$
	be a quasi-subsequential $k$-separated finite-state transducer. The
	generalised function of transitions $\delta^* : \prod_{i=1}^k{Q_i}
	\times \Sigma^* \rightarrow \prod_{i=1}^k{Q_i}$ is defined as the least
	function such that $\delta^*(q, \epsilon) = q$ for $q \in
	\prod_{i=1}^k{Q_i}$ and $\delta^*(q, w\sigma) = \delta(\delta^*(q, w),
	\sigma)$ for $q \in \prod_{i=1}^k{Q_i}$, $w \in \Sigma^*$ and $\sigma \in
	\Sigma$.
\end{definition}
