\subsection{Finite-state transducers}

\begin{definition}
	A finite-state transducer $\mathcal{T}$ is a tuple $\langle \Sigma, \Omega,
	Q, S, F, \Delta \rangle$ where $\Sigma$ and $\Omega$ are alphabets, $Q$ is
	a finite set of states, $S \subseteq Q$ is a set of start states, $F
	\subseteq Q$ is a set of final states and $\Delta : Q \times (\Sigma
	\times \Omega^*) \times Q$ is a relation of transitions and outputs.
\end{definition}

\begin{definition}
	A path $\pi$ in a finite-state transducer $\mathcal{T} = \langle \Sigma,
	\Omega, Q, S, F, \Delta \rangle$ is a non-empty sequence $\langle q_1,
	\sigma_1, w_1, q_2 \rangle, \langle q_2, \sigma_2, w_2, q_3 \rangle, \ldots,
	\langle q_{n-1}, \sigma_{n-1}, w_{n-1}, q_n \rangle$ where $\forall i \in
	\interval{1}{n}(q_i \in Q)$, $\forall i \in \interval{1}{n-1}(\sigma_i
	\in \Sigma)$, $\forall i \in \interval{1}{n-1}(w_i \in \Omega^*)$ and
	$\forall i \in \interval{1}{n-1}(\langle q_i, \sigma_i, w_i, q_{i+1} \rangle
	\in \Delta)$. The path $\pi$ is successful if $q_1 \in S$ and $q_n \in F$.
	The label of the path $\pi$ is the word $\sigma_1\sigma_2\ldots\sigma_{n-1}$
	of length $n-1$ over $\Sigma$. The output of the path $\pi$ is the word
	$w_1 w_2 \ldots w_{n-1}$ over $\Omega$.
\end{definition}

\begin{definition}
	Let $\mathcal{T} = \langle \Sigma, \Omega, Q, S, F, \Delta \rangle$ be a
	finite-state transducer. The generalized relation of transitions and outputs
	$\Delta^* : Q \times (\Sigma^* \times \Omega^*) \times Q$ is defined as
	the least relation such that $\forall q_1 \in Q \forall \sigma \in
	\Sigma \forall w \in \Omega^* (\langle q_1, \sigma, w, q_2 \rangle \in
	\Delta \implies \langle q_1, \sigma, w, q_2 \rangle \in \Delta^*)$ and
	$\forall q_1 \in Q \forall w \in \Sigma^* \forall \sigma \in \Sigma
	\forall u, v \in \Omega^* (\langle q_1, w, u, q_2 \rangle \in \Delta^*
	\land \langle q_2, \sigma, v, q_3 \rangle \in \Delta \implies
	\langle q_1, w\sigma, uv, q_3 \rangle \in \Delta^*)$.
\end{definition}

\begin{definition}
	The relation recognised by a finite-state transducer $\mathcal{T} =
	\langle \Sigma, \Omega, Q, S, F, \Delta \rangle$ is the set of all pairs
	of labels and outputs of successful paths $\mathcal{R}(\mathcal{T}) = \{
	\langle u, v \rangle \in \Sigma^* \times \Omega^* \mid \exists i \in S
	\exists f \in F (\langle i, u, v, f \rangle \in \Delta^*) \}$.
\end{definition}

\begin{definition}
	A subsequential finite-state transducer $\mathcal{T}$ is a tuple $\langle
	\Sigma, \Omega, Q, s,$\\
	$F, \delta, \lambda, \Phi \rangle$ where $\Sigma$ and $\Omega$ are
	alphabets, $Q$ is a finite set of states, $s \in Q$ is a start state,
	$F \subseteq Q$ is a set of final states, $\delta : Q \times \Sigma
	\rightarrow Q$ is a transition function, $\lambda : Q \times \Sigma
	\rightarrow \Omega^*$ is a transition output function and $\Phi : F
	\rightarrow \Omega^*$ is a final state output function.
\end{definition}

\begin{definition}
	A path $\pi$ in a subsequential finite-state transducer $\mathcal{T} =
	\langle \Sigma, \Omega, Q, s, F, \delta, \lambda, \Phi \rangle$ is a
	non-empty sequence $\langle q_1, \sigma_1, w_1, q_2 \rangle, \langle q_2,
	\sigma_2, w_2, q_3 \rangle, \ldots,$\\
	$\langle q_{n-1}, \sigma_{n-1}, w_{n-1},
	q_n \rangle$ where $\forall i \in \interval{1}{n}(q_i \in Q)$, $\forall i
	\in \interval{1}{n-1}(\sigma_i \in \Sigma)$, $\forall i \in
	\interval{1}{n-1}(w_i \in \Omega^*)$, $\forall i \in \interval{1}{n-1}
	(q_{i+1} = \delta(q_i, \sigma_i))$ and $\forall i \in \interval{1}{n-1}
	(w_i = \lambda(q_i, \sigma_i))$. The path $\pi$ is successful if $q_1 \in
	S$ and $q_n \in F$. The label of the path $\pi$ is the word
	$\sigma_1\sigma_2\ldots\sigma_{n-1}$ of length $n-1$ over $\Sigma$. The
	output of the path $\pi$ is the word $w_1 w_2 \ldots w_{n-1} \Phi(q_n)$
	over $\Omega$.
\end{definition}

\begin{definition}
	Let $\mathcal{T} = \langle \Sigma, \Omega, Q, s, F, \delta, \lambda, \Phi
	\rangle$ be a finite-state transducer. The generalized function of
	transitions $\delta^* : Q \times \Sigma^* \times Q$ is defined as the least
	function such that $\forall q_1 \in Q \forall \sigma \in \Sigma
	(\delta(q_1, \sigma) = q_2 \implies \delta^*(q_1, \sigma) = q_2)$ and
	$\forall q_1 \in Q \forall w \in \Sigma^* \forall \sigma \in \Sigma
	(\delta^*(q_1, w) = q_2 \land \delta(q_2, \sigma) = q_3 \implies
	\delta^*(q_1, w\sigma) = q_3)$.
\end{definition}

\begin{theorem}[Choffrut, Mohri \cite{C77, M96}]
	Let $\mathcal{T}$ be a finite-state transducer. We can construct a
	subsequential finite-state transducer $\mathcal{T'}$ such that $\mathcal{R}
	( \mathcal{T'} ) = \mathcal{R} ( \mathcal{T} )$.
\end{theorem}
