\subsection{Finite-state transducers}

\begin{definition}
	A finite-state transducer $\mathcal{T}$ is a tuple
	\begin{align*}
		\langle \Sigma, \Omega, Q, S, F, \Delta \rangle
	\end{align*}
	where
	\begin{itemize}
		\item $\Sigma$ and $\Omega$ are respectively input and output
			alphabets,
		\item $Q$ is a finite set of states,
		\item $S \subseteq Q$ is a set of start states,
		\item $F \subseteq Q$ is a set of final states,
		\item $\Delta : Q \times (\Sigma_\epsilon \times \Omega^*) \times
			Q$ is a relation of state transitions and outputs.
	\end{itemize}
\end{definition}

\begin{definition}
	A path $\pi$ in a finite-state transducer $\mathcal{T} = \langle \Sigma,
	\Omega, Q, S, F, \Delta \rangle$ is a non-empty sequence of state
	transitions and outputs
	\begin{align*}
		\langle q_1, \sigma_1, w_1, q_2 \rangle, \langle q_2, \sigma_2, w_2,
		q_3 \rangle, \ldots, \langle q_{n-1}, \sigma_{n-1}, w_{n-1}, q_n
		\rangle
	\end{align*}
	where
	\begin{itemize}
		\item $q_i \in Q$ for $i \in \interval{1}{n}$,
		\item $\sigma_i \in \Sigma_\epsilon$ for $i \in \interval{1}{n-1}$,
		\item $w_i \in \Omega^*$ for $i \in \interval{1}{n-1}$,
		\item $\langle q_i, \sigma_i, w_i, q_{i+1} \rangle \in \Delta$ for
			$i \in \interval{1}{n-1}$.
	\end{itemize}
	The path $\pi$ is successful if and only if $q_1 \in S$ and $q_n \in F$. The
	label of the path $\pi$ is the word $\sigma_1\sigma_2\ldots\sigma_{n-1}$
	over $\Sigma$. The output of the path $\pi$ is the word $w_1 w_2
	\ldots w_{n-1}$ over $\Omega$.
\end{definition}

\begin{definition}
	Let $\mathcal{T} = \langle \Sigma, \Omega, Q, S, F, \Delta \rangle$ be a
	finite-state transducer. The generalized relation of state transitions and
	outputs $\Delta^* : Q \times (\Sigma^* \times \Omega^*) \times Q$ is defined
	inductively as the least relation such that
	\begin{itemize}
		\item $\langle q, \epsilon, \epsilon, q \rangle \in \Delta^*$ for
			$q \in Q$,
		\item $\langle q', w\sigma, uv, q''' \rangle \in \Delta^*$ for
			$\langle q', w, u, q'' \rangle \in \Delta^*$ and $\langle
			q'', \sigma, v, q''' \rangle \in \Delta$.
	\end{itemize}
\end{definition}

\begin{definition}
	Let $\mathcal{T} = \langle \Sigma, \Omega, Q, S, F, \Delta \rangle$ be a
	finite-state transducer. The $\epsilon$-closure of a state of $\mathcal{T}$
	is the function $\Delta_\epsilon : Q \rightarrow 2^{Q \times \Omega^*}$
	defined as
	\begin{align*}
		\Delta_\epsilon(q) = \{ \langle p, u \rangle \in Q \times \Omega^*
			\mid \exists \langle q, \epsilon, u, p \rangle \in \Delta^*
			\}.
	\end{align*}
\end{definition}

\begin{definition}
	Let $\mathcal{T} = \langle \Sigma, \Omega, Q, S, F, \Delta \rangle$ be a
	finite-state transducer. The longest common output function
	$\Delta_\wedge : \Sigma^* \rightarrow \Omega^*$ is defined as
	\begin{align*}
		\Delta_\wedge(w) = \bigwedge{\{ u \in \Omega^* \mid \exists s \in S
		\; \exists q \in Q (\langle s, w, u, q \rangle \in \Delta^*) \}}.
	\end{align*}
\end{definition}

\begin{definition}
	The relation recognized by a finite-state transducer $\mathcal{T} =
	\langle \Sigma, \Omega, Q, S, F, \Delta \rangle$ is the set of all pairs
	of labels and outputs of successful paths
	\begin{align*}
		\mathcal{R}_\mathcal{T} = \{ \langle u, v \rangle \in \Sigma^*
		\times \Omega^* \mid \exists s \in S \; \exists f \in F (\langle s,
		u, v, f \rangle \in \Delta^*) \}.
	\end{align*}
\end{definition}

\begin{definition}
	A finite-state transducer $\mathcal{T} = \langle \Sigma, \Omega, Q, S, F,
	\Delta \rangle$ is trimmed if and only if for every state $q \in Q$ there
	exist $s \in S$, $f \in F$, $w', w'' \in \Sigma^*$ and $v', v'' \in
	\Omega^*$ such that $\langle s, w', v', q \rangle \in \Delta^*$ and
	$\langle q, w'', v'', f \rangle \in \Delta^*$.
\end{definition}

\begin{proposition}[see \cite{MS19}]
	Let $\mathcal{T}$ be a finite-state transducer. There exists a trimmed
	finite-state transducer $\mathcal{T'}$ such that $\mathcal{R}_\mathcal{T}
	= \mathcal{R}_\mathcal{T'}$.
\end{proposition}

\begin{definition}
	A finite-state transducer $\mathcal{T}$ is functional if and only if
	$\mathcal{R}_\mathcal{T}$ is a function. If $\mathcal{T}$ is a functional
	finite-state transducer we shall write $\mathcal{F}_\mathcal{T}$ instead of
	$\mathcal{R}_\mathcal{T}$.
\end{definition}

\begin{definition}
	The graph $\mathcal{G}_{\mathcal{T}}$ corresponding to a finite-state
	transducer $\mathcal{T} = \langle \Sigma, \Omega, Q, S, F, \Delta \rangle$
	is the triple
	\begin{align*}
		\langle Q, \Sigma_\epsilon \times \Omega^*, \Delta \rangle.
	\end{align*}
\end{definition}

\begin{definition}
	A finite-state transducer $\mathcal{T}$ is planar if and only if its
	corresponding graph $\mathcal{G}_{\mathcal{T}}$ is planar.
\end{definition}

\begin{definition}
	A subsequential finite-state transducer $\mathcal{T}$ is a tuple
	\begin{align*}
		\langle \Sigma, \Omega, Q, s, F, \delta, \lambda, \psi \rangle
	\end{align*}
	where
	\begin{itemize}
		\item $\Sigma$ and $\Omega$ are respectively input and output
			alphabets,
		\item $Q$ is a finite set of states,
		\item $s \in Q$ is a start state, $F \subseteq Q$ is a set of final
			states,
		\item $\delta : Q \times \Sigma \rightarrow Q$ is a state transition
			function,
		\item $\lambda : Q \times \Sigma \rightarrow \Omega^*$ is a
			state transition output function,
		\item $\psi : F \rightarrow \Omega^*$ is a final state output
			function.
	\end{itemize}
\end{definition}

\begin{definition}
	A path $\pi$ in a subsequential finite-state transducer $\mathcal{T} =
	\langle \Sigma, \Omega, Q, s, F, \delta, \lambda, \psi \rangle$ is a
	non-empty sequence of state transitions
	\begin{align*}
		\langle q_1, \sigma_1, w_1, q_2 \rangle, \langle q_2, \sigma_2, w_2,
		q_3 \rangle, \ldots, \langle q_{n-1}, \sigma_{n-1}, w_{n-1}, q_n
		\rangle
	\end{align*}
	where
	\begin{itemize}
		\item $q_i \in Q$ for $i \in \interval{1}{n}$,
		\item $\sigma_i \in \Sigma$ for $i \in \interval{1}{n-1}$,
		\item $w_i = \lambda(q_i, \sigma_i)$ for $i \in \interval{1}{n-1}$,
		\item $q_{i+1} = \delta(q_i, \sigma_i)$ for $i \in
			\interval{1}{n-1}$.
	\end{itemize}
	The path $\pi$ is successful if and only if $q_1 = s$ and $q_n \in F$. The
	label of the path $\pi$ is the word $\sigma_1\sigma_2\ldots\sigma_{n-1}$ of
	length $n-1$ over $\Sigma$. The output of the path $\pi$ is the word $w_1
	w_2 \ldots w_{n-1} \psi(q_n)$ over $\Omega$.
\end{definition}

\begin{definition}
	Let $\mathcal{T} = \langle \Sigma, \Omega, Q, s, F, \delta, \lambda, \psi
	\rangle$ be a subsequential finite-state transducer. The generalized
	function of state transitions $\delta^* : Q \times \Sigma^* \rightarrow Q$
	is defined inductively as the least function such that
	\begin{itemize}
		\item $\delta^*(q, \epsilon) = q$ for $q \in Q$,
		\item $\delta^*(q, w\sigma) = \delta(\delta^*(q, w), \sigma)$ for
			$q \in Q$, $w \in \Sigma^*$ and $\sigma \in \Sigma$.
	\end{itemize}
\end{definition}

\begin{definition}
	Let $\mathcal{T} = \langle \Sigma, \Omega, Q, s, F, \delta, \lambda, \psi
	\rangle$ be a subsequential finite-state transducer. The generalized
	function of state transition outputs $\lambda^* : Q \times \Sigma^*
	\rightarrow \Omega^*$ is defined inductively as the least function such that
	\begin{itemize}
		\item $\lambda^*(q, \epsilon) = \epsilon$ for $q \in Q$,
		\item $\lambda^*(q, w\sigma) = \lambda^*(q, w) \lambda(\delta^*(q,
			w), \sigma)$ for $q \in Q$, $w \in \Sigma^*$ and $\sigma
			\in \Sigma$.
	\end{itemize}
\end{definition}

\begin{definition}
	The function recognized by a subsequential finite-state transducer
	$\mathcal{T} = \langle \Sigma, \Omega, Q, s, F, \delta, \lambda, \psi
	\rangle$ is the set of all pairs of labels and outputs of successful paths
	\begin{align*}
		\mathcal{F}_\mathcal{T} = \{ \langle w, \lambda^*(w)\psi(f) \rangle
		\in \Sigma^* \times \Omega^* \mid \exists f \in F (\delta^*(s, w) =
		f) \}.
	\end{align*}
\end{definition}

\begin{theorem}[Mohri, see \cite{M96}]\label{t_choffrut_mohri}
	Let $\mathcal{T}$ be a functional finite-state transducer and let
	$\mathcal{F}_\mathcal{T}$ have bounded variation. There exists a
	subsequential finite-state transducer $\mathcal{T'}$ such that
	$\mathcal{F}_\mathcal{T'} = \mathcal{F}_\mathcal{T}$.
\end{theorem}
