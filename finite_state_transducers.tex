\subsection{Finite-state transducers}

\begin{definition}
	A finite-state transducer $\mathcal{T}$ is a tuple
	\begin{align*}
		\langle \Sigma, \Omega, Q, S, F, \Delta \rangle
	\end{align*}
	where $\Sigma$ and $\Omega$ are alphabets, $Q$ is a finite set of states,
	$S \subseteq Q$ is a set of start states, $F \subseteq Q$ is a set of final
	states and $\Delta : Q \times (\Sigma_\epsilon \times \Omega^*) \rightarrow
	2^Q$ is a function of transitions and outputs.
\end{definition}

\begin{definition}
	A path $\pi$ in a finite-state transducer $\mathcal{T} = \langle \Sigma,
	\Omega, Q, S, F, \Delta \rangle$ is a non-empty sequence of transitions
	\begin{align*}
		\langle q_1, \sigma_1, w_1, q_2 \rangle, \langle q_2, \sigma_2, w_2,
		q_3 \rangle, \ldots, \langle q_{n-1}, \sigma_{n-1}, w_{n-1}, q_n
		\rangle
	\end{align*}
	where $\forall i \in \interval{1}{n}(q_i \in Q)$, $\forall i \in
	\interval{1}{n-1}(\sigma_i \in \Sigma_\epsilon)$, $\forall i \in
	\interval{1}{n-1}(w_i \in \Omega^*)$ and $\forall i \in
	\interval{1}{n-1}(q_{i+1} \in \Delta(q_i, \sigma_i, w_i))$. The path $\pi$
	is successful if and only if $q_1 \in S$ and $q_n \in F$. The label of the
	path $\pi$ is the word $\sigma_1\sigma_2\ldots\sigma_{n-1}$ over $\Sigma$.
	The output of the path $\pi$ is the word $w_1 w_2 \ldots w_{n-1}$ over
	$\Omega$.
\end{definition}

\begin{definition}
	Let $\mathcal{T} = \langle \Sigma, \Omega, Q, S, F, \Delta \rangle$ be a
	finite-state transducer. The generalized function of transitions and outputs
	$\Delta^* : Q \times (\Sigma^* \times \Omega^*) \rightarrow 2^Q$ is defined
	inductively as the least function such that $\Delta^*(q, \epsilon, \epsilon)
	= \{ q \}$ for $q \in Q$ and $\Delta^*(q, w\sigma, uv) = \Delta(\Delta^*(
	q, w, u), \sigma, v)$ for $q \in Q$, $w \in \Sigma^*$, $\sigma \in
	\Sigma_\epsilon$ and $u, v \in \Omega^*$.
\end{definition}

\begin{definition}
	The relation recognized by a finite-state transducer $\mathcal{T} =
	\langle \Sigma, \Omega, Q, S, F, \Delta \rangle$ is the set of all pairs
	of labels and outputs of successful paths
	\begin{align*}
		\mathcal{R}(\mathcal{T}) = \{ \langle u, v \rangle \in \Sigma^*
		\times \Omega^* \mid \hat{\Delta}^* (S, \{ u \}, \{ v \}) \cap F
		\neq \emptyset \}.
	\end{align*}
\end{definition}

\begin{definition}
	A finite-state transducer $\mathcal{T}$ is functional if and only if
	$\mathcal{R}(\mathcal{T})$ is a function.
\end{definition}

\begin{definition}
	The graph $\mathcal{G}_{\mathcal{T}}$ corresponding to a finite-state
	transducer $\mathcal{T} = \langle \Sigma, \Omega, Q, S, F, \Delta \rangle$
	is the triple
	\begin{align*}
		\langle Q, \Sigma_\epsilon \times \Omega^*, \Delta \rangle.
	\end{align*}
\end{definition}

\begin{definition}
	A finite-state transducer $\mathcal{T}$ is planar if and only if its
	corresponding graph $\mathcal{G}_{\mathcal{T}}$ is planar.
\end{definition}

\begin{definition}
	A subsequential finite-state transducer $\mathcal{T}$ is a tuple
	\begin{align*}
		\langle \Sigma, \Omega, Q, s, F, \delta, \lambda, \Phi \rangle
	\end{align*}
	where $\Sigma$ and $\Omega$ are alphabets, $Q$ is a finite set of states,
	$s \in Q$ is a start state, $F \subseteq Q$ is a set of final states,
	$\delta : Q \times \Sigma \rightarrow Q$ is a transition function, $\lambda
	: Q \times \Sigma \rightarrow \Omega^*$ is a transition output function and
	$\Phi : F \rightarrow \Omega^*$ is a final state output function.
\end{definition}

\begin{definition}
	A path $\pi$ in a subsequential finite-state transducer $\mathcal{T} =
	\langle \Sigma, \Omega, Q, s, F, \delta, \lambda, \Phi \rangle$ is a
	non-empty sequence of transitions
	\begin{align*}
		\langle q_1, \sigma_1, w_1, q_2 \rangle, \langle q_2, \sigma_2, w_2,
		q_3 \rangle, \ldots, \langle q_{n-1}, \sigma_{n-1}, w_{n-1}, q_n
		\rangle
	\end{align*}
	where $\forall i \in \interval{1}{n}(q_i \in Q)$, $\forall i \in
	\interval{1}{n-1}(\sigma_i \in \Sigma)$, $\forall i \in
	\interval{1}{n-1}(w_i \in \Omega^*)$, $\forall i \in \interval{1}{n-1}
	(q_{i+1} = \delta(q_i, \sigma_i))$ and $\forall i \in \interval{1}{n-1}
	(w_i = \lambda(q_i, \sigma_i))$. The path $\pi$ is successful if and only
	if $q_1 \in S$ and $q_n \in F$. The label of the path $\pi$ is the word
	$\sigma_1\sigma_2\ldots\sigma_{n-1}$ of length $n-1$ over $\Sigma$. The
	output of the path $\pi$ is the word $w_1 w_2 \ldots w_{n-1} \Phi(q_n)$
	over $\Omega$.
\end{definition}

\begin{definition}
	Let $\mathcal{T} = \langle \Sigma, \Omega, Q, s, F, \delta, \lambda, \Phi
	\rangle$ be a subsequential finite-state transducer. The generalized
	function of transitions $\delta^* : Q \times \Sigma^* \rightarrow Q$ is
	defined inductively as the least function such that $\delta^*(q, \epsilon)
	= q$ for $q \in Q$ and $\delta^*(q, w\sigma) = \delta(\delta^*(q, w),
	\sigma)$ for $q \in Q$, $w \in \Sigma^*$ and $\sigma \in \Sigma$.
\end{definition}

\begin{definition}
	Let $\mathcal{T} = \langle \Sigma, \Omega, Q, s, F, \delta, \lambda, \Phi
	\rangle$ be a subsequential finite-state transducer. The generalized
	function of transition outputs $\lambda^* : Q \times \Sigma^* \rightarrow
	\Omega^*$ is defined inductively as the least function such that
	$\lambda^*(q, \epsilon) = \epsilon$ for $q \in Q$ and $\lambda^*(q, w\sigma)
	= \lambda^*(q, w) \lambda(\delta^*(q, w), \sigma)$ for $q \in Q$, $w \in
	\Sigma^*$ and $\sigma \in \Sigma$.
\end{definition}

\begin{definition}
	The relation recognized by a subsequential finite-state transducer
	$\mathcal{T} = \langle \Sigma, \Omega, Q, s, F, \delta, \lambda, \Phi
	\rangle$ is the set of all pairs of labels and outputs of successful paths
	\begin{align*}
		\mathcal{R}(\mathcal{T}) = \{ \langle w, \lambda^*(w)\Phi(f) \rangle
		\in \Sigma^* \times \Omega^* \mid \exists f \in F (\delta^*(s, w) =
		f) \}.
	\end{align*}
\end{definition}

\begin{theorem}[Choffrut, Mohri \cite{C77, M96}]\label{t_choffrut_mohri}
	Let $\mathcal{T} = \langle \Sigma, \Omega, Q, S, F, \Delta \rangle$ be a
	functional finite-state transducer. We can construct a subsequential
	finite-state transducer $\mathcal{T'} = \langle \Sigma, \Omega, Q', s', F',
	\delta, \lambda, \Phi \rangle$ such that $\mathcal{R}(\mathcal{T'}) =
	\mathcal{R}(\mathcal{T})$.
\end{theorem}
