\subsection{Regular expressions}

\begin{definition}
	Let $\Sigma$ be an alphabet such that $\Sigma \cap \{ \varnothing, (, ), .,
	+, * \} = \emptyset$. The class of regular expressions over $\Sigma$ and the
	languages they denote is defined as the inclusion-wise least subset of
	$(\Sigma \cup \{
	\varnothing, (, ), ., +, * \})^*$ such that
	\begin{enumerate}
		\item $\varnothing$ is a regular expression denoting the language
			$\emptyset$,
		\item if $\sigma \in \Sigma_\epsilon$ then $\sigma$ is a regular
			expression denoting the language $\{ \sigma \}$,
		\item if $\mathcal{R}_1$ and $\mathcal{R}_2$ are regular expressions
			denoting the languages $\mathcal{L}_{\mathcal{R}_1}$
			and $\mathcal{L}_{\mathcal{R}_2}$ respectively then
			\begin{itemize}
				\item $(\mathcal{R}_1.\mathcal{R}_2)$ is a regular
					expression denoting the language
					$\mathcal{L}_{\mathcal{R}_1} \hat{\cdot}
					\mathcal{L}_{\mathcal{R}_2}$,
				\item $(\mathcal{R}_1+\mathcal{R}_2)$ is a regular
					expression denoting the language
					$\mathcal{L}_{\mathcal{R}_1} \cup
					\mathcal{L}_{\mathcal{R}_2}$,
				\item $(\mathcal{R}_1*)$ is a regular expression
					denoting the language
					$\mathcal{L}_{\mathcal{R}_1}^*$.
			\end{itemize}
	\end{enumerate}
\end{definition}

\begin{definition}
	Let $\mathcal{R}$ be a regular expression over an alphabet $\Sigma$. We
	write $\mathcal{L}_\mathcal{R}$ for the language denoted by $\mathcal{R}$.
\end{definition}

\begin{theorem}[Thompson, see \cite{T68}]\label{t_thompson}
	Let $\mathcal{R}$ be a regular expression over an alphabet $\Sigma$. A
	finite-state automaton $\mathcal{A}$ such that $\mathcal{L}_\mathcal{A} =
	\mathcal{L}_\mathcal{R}$ can be constructed from $\mathcal{R}$ as follows.
	\begin{enumerate}
		\item If $\mathcal{R} = \varnothing$ then $\mathcal{A} = \langle
			\Sigma, \{ s, f \}, \{ s \}, \{ f \}, \emptyset \rangle$ is
			the Type 1 Thompson finite-state automaton constructed from
			$\mathcal{R}$,
		\item if $\mathcal{R} = \sigma$ for $\sigma \in \Sigma_\epsilon$
			then $\mathcal{A} = \langle \Sigma, \{ s, f \}, \{ s \}, \{
			f \}, \{ \langle s, \sigma, f \rangle \} \rangle$ is the Type
			2 Thompson finite-state automaton constructed from
			$\mathcal{R}$,
		\item if $\mathcal{A}' = \langle \Sigma, Q', S', F', \Delta'
			\rangle$ and $\mathcal{A}'' = \langle \Sigma, Q'', S'', F'',
			\Delta'' \rangle$ are the Thompson finite-state automata
			constructed from $\mathcal{R}'$ and $\mathcal{R}''$
			respectively then
			\begin{itemize}
				\item $\mathcal{A} = \langle \Sigma, Q' \cup Q'',
					S', F'', \Delta \rangle$ is the Type 3
					Thompson finite-state automaton constructed
					from $(\mathcal{R}'.\mathcal{R}'')$ where
					$\Delta = \Delta' \cup \Delta'' \cup \{
					\langle f', \epsilon, s'' \rangle \mid
					\exists f' \in F' \; \exists s'' \in S'' \}$,
				\item $\mathcal{A} = \langle \Sigma, Q' \cup Q''
					\cup \{ s, f \}, \{ s \}, \{ f \},
					\Delta \rangle$ is the Type 4 Thompson
					finite-state automaton constructed from
					$(\mathcal{R}'+\mathcal{R}'')$ where 
					$\Delta = \Delta' \cup \Delta'' \cup \{$
					$\langle s, \epsilon, s' \rangle$,
					$\langle s, \epsilon, s'' \rangle$,
					$\langle f', \epsilon, f \rangle$,
					$\langle f'', \epsilon, f \rangle \mid
					\exists s' \in S' \; \exists s'' \in S'' \;
					\exists f' \in F' \; \exists f'' \in F'' \}$,
				\item $\mathcal{A} = \langle \Sigma, Q' \cup \{ s,
					f \}, \{ s \}, \{ f \}, \Delta \rangle$
					is the Type 5 Thompson finite-state automaton
					constructed from $(\mathcal{R}'*)$ where 
					$\Delta = \Delta' \cup \{$
					$\langle s, \epsilon, s' \rangle$,
					$\langle s, \epsilon, f \rangle$,
					$\langle f', \epsilon, f \rangle$,
					$\langle f', \epsilon, s' \rangle \mid
					\exists s' \in S' \; \exists f' \in F' \}$.
			\end{itemize}
	\end{enumerate}
\end{theorem}

\begin{proposition}
	Let $\mathcal{A} = \langle \Sigma, Q, S, F, \Delta \rangle$ be a Thompson
	finite-state automaton. Then $|S| = |F| = 1$.
\end{proposition}

\begin{proof}
	We prove that $|S| = |F| = 1$ by induction on the construction of the
	Thompson finite-state automaton.

	\begin{case}[$\mathcal{A}$ is Type 1, 2, 4 or 5]
		$|S| = |\{ s \}| = 1$ and $|F| = |\{ f \}| = 1$ by construction.
	\end{case}

	\begin{case}[$\mathcal{A}$ is Type 3]
		$|S| = |S'| = 1$ and $|F| = |F'| = 1$ by the inductive hypothesis
		for $\mathcal{A'}$ and $\mathcal{A''}$.
	\end{case}
\end{proof}
