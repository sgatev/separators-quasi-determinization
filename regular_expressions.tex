\subsection{Regular expressions}

\begin{definition}
	Let $\Sigma$ be an alphabet such that $\Sigma \cap \{ \varnothing, (, ), .,
	+, * \} = \emptyset$. The class of regular expressions over $\Sigma$ and the
	languages they denote is defined recursively as the least set such that
	\begin{enumerate}
		\item $\varnothing$ is a regular expression denoting the language
			$\emptyset$,
		\item if $\sigma \in \Sigma_\epsilon$ then $\sigma$ is a regular
			expression denoting the language $\{ \sigma \}$,
		\item if $\mathcal{R}_1$ and $\mathcal{R}_2$ are regular expressions
			denoting the languages $\mathcal{L}_{\mathcal{R}_1}$
			and $\mathcal{L}_{\mathcal{R}_2}$ respectively then
			\begin{itemize}
				\item $(\mathcal{R}_1.\mathcal{R}_2)$ is a regular
					expression denoting the language
					$\mathcal{L}_{\mathcal{R}_1} \hat{\cdot}
					\mathcal{L}_{\mathcal{R}_2}$,
				\item $(\mathcal{R}_1+\mathcal{R}_2)$ is a regular
					expression denoting the language
					$\mathcal{L}_{\mathcal{R}_1} \cup
					\mathcal{L}_{\mathcal{R}_2}$,
				\item $(\mathcal{R}_1*)$ is a regular expression
					denoting the language
					$\mathcal{L}_{\mathcal{R}_1}^*$.
			\end{itemize}
	\end{enumerate}
\end{definition}

\begin{definition}
	Let $\mathcal{R}$ be a regular expression over an alphabet $\Sigma$. We
	write $\mathcal{L}_\mathcal{R}$ for the language denoted by $\mathcal{R}$.
\end{definition}

\begin{theorem}[Thompson, see \cite{T68}]\label{t_thompson}
	Let $\mathcal{R}$ be a regular expression over an alphabet $\Sigma$. There
	exists a finite-state automaton $\mathcal{A}$ such that
	$\mathcal{L}_\mathcal{A} = \mathcal{L}_\mathcal{R}$.
\end{theorem}
