\subsection{Regular expressions}

\begin{definition}
	A regular expression $\mathcal{R}$ over an alphabet $\Sigma$ such that
	$\Sigma \cap \{ \varnothing, (, ), ., +, * \} = \emptyset$ is a word over
	$\Sigma \cup \{ \varnothing, (, ), ., +, * \}$ denoting a regular language
	$\mathcal{L}(\mathcal{R})$ over $\Sigma$ defined by induction:
	\begin{enumerate}
		\item $\varnothing$ is a regular expression over $\Sigma$
			denoting the regular language $\emptyset$;
		\item if $\sigma \in \Sigma_\epsilon$ then $\sigma$ is a regular
			expression over $\Sigma$ denoting the regular language $\{
			\sigma \}$;
		\item if $\mathcal{R}'$ and $\mathcal{R}''$ are regular expressions
			over $\Sigma$ denoting the regular languages $\mathcal{L}
			(\mathcal{R}')$ and $\mathcal{L}(\mathcal{R}'')$
			respectively then
			\begin{itemize}
				\item $(\mathcal{R}'.\mathcal{R}'')$ is a regular
					expression over $\Sigma$ denoting the
					regular language $\mathcal{L}(\mathcal{R}')
					\hat{\cdot} \mathcal{L}(\mathcal{R}'')$;
				\item $(\mathcal{R}'+\mathcal{R}'')$ is a regular
					expression over $\Sigma$ denoting the
					regular language $\mathcal{L}(\mathcal{R}')
					\cup \mathcal{L}(\mathcal{R}'')$;
				\item $(\mathcal{R}'*)$ is a regular
					expression over $\Sigma$ denoting the
					regular language $\mathcal{L}
					(\mathcal{R}')^*$;
			\end{itemize}
		\item there are no other regular expressions over $\Sigma$.
	\end{enumerate}
\end{definition}

\begin{theorem}[Thompson, see \cite{T68}]\label{t_thompson}
	Let $\mathcal{R}$ be a regular expression over an alphabet $\Sigma$. There
	exists a finite-state automaton $\mathcal{A}$ such that
	$\mathcal{L}(\mathcal{A}) = \mathcal{L}(\mathcal{R})$.
\end{theorem}
