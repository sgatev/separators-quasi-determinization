\subsection{Sets, relations and functions}

\begin{definition}
	A set is defined as usual. We denote that an object $x$ belongs
	to a set $S$ by writing $x \in S$. Alternatively, we write $x
	\notin S$ to denote that $x$ does not belong to $S$. We write $S_1
	\cup S_2$ for the union of sets the $S_1$ and $S_2$, and $S_1 \cap
	S_2$ for their intersection. The cardinality of a set $S$ is denoted
	by $\lvert S \rvert$.
\end{definition}

\begin{definition}
	A tuple is defined as usual. We write $\langle x_1, x_2, \dots, x_n
	\rangle$ to denote the tuple of the $n$ objects $x_1, x_2, \dots, x_n$.
\end{definition}

\begin{definition}
	The cartesian product of the sets $S_1, S_2, \dots, S_n$ is defined
	as the set $\prod_{i=1}^n{S_i} = \{\langle s_1, s_2, \dots, s_n \rangle
	\lvert s_i \in S_i$ for $1 \leq i \leq n\}$.
\end{definition}

\begin{definition}
	An $n$-ary relation $R$ on the sets $S_1, S_2, \dots, S_n$ is a subset
	$R \subseteq \prod_{i=1}^n{S_i}$.
\end{definition}

\begin{definition}
	A binary relation $R \subseteq M \times N$ is a function if $\forall
	\langle m', n' \rangle, \langle m'', n'' \rangle \in R (n' \neq n''
	\implies m' \neq m'')$.
\end{definition}
