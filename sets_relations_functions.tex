\subsection{Sets, relations and functions}

Sets, relations, functions and tuples are introduced as usual. The cardinality of a
set $S$ is indicated by $|S|$. The unique set with no members is denoted by
$\emptyset$ and $2^S$ stands for the powerset of a set $S$. The set of natural
numbers is indicated by $\mathbb{N}$ and $\{ m, m+1, \ldots, n \}$ denotes the set
of all natural numbers in the interval $\interval{m}{n}$ for $m, n \in \mathbb{N}$.
A tuple $t$ of $n$ elements is denoted by $\langle t_1, t_2, \ldots, t_n \rangle$
and its length is indicated by $|t| = n$.

\begin{definition}
	The sets $S_1, S_2, \ldots, S_n$ form a partition of the set $S$ for $n > 0$
	if and only if
	\begin{itemize}
		\item $S = \bigcup_{i=1}^n{S_i}$,
		\item $|S_i| > 0$ for $i \in \{ 1, 2, \ldots, n \}$,
		\item $S_i \cap S_j = \emptyset$ if $i \neq j$ for $i, j \in
			\{ 1, 2, \ldots, n \}$.
	\end{itemize}
	We shall write $P_S = \{ S_1, S_2, \ldots, S_n \}$ to indicate that the sets
	$S_1, S_2, \ldots, S_n$ form a partition of the set $S$.
\end{definition}

\begin{definition}
	Let $f: X \rightarrow Y$ be a function. The domain and range of $f$ are the
	sets respectively defined as
	\begin{align*}
		dom(f) &= \{ x \in X \mid \exists y \in Y (f(x) = y) \},\\
		rng(f) &= \{ y \in Y \mid \exists x \in X (f(x) = y) \}.
	\end{align*}
\end{definition}

\begin{definition}
	A function $f: X \rightarrow Y$ is injective if and only if $x' \neq x''$
	implies $f(x') \neq f(x'')$ for $x', x'' \in dom(f)$.
\end{definition}

\begin{definition}
	Let $f: X \rightarrow Y$ be a function. The set-lifted function $\hat{f}
	: 2^X \rightarrow 2^Y$ is defined as
	\begin{align*}
		\hat{f}(X') = \{ f(x) \in Y \mid \exists x \in X' \cap dom(f) \}.
	\end{align*}
\end{definition}

\begin{definition}
	Let $f: \mathbb{N} \rightarrow \mathbb{R}$ and $g: \mathbb{N} \rightarrow
	\mathbb{R}$ be functions. We shall write $f \in \mathcal{O}(g)$ if and only
	if there exist $c > 0$ and $m \in \mathbb{N}$ such that $|f(n)| \leq
	c|g(n)|$ for every $n > m$.
\end{definition}
