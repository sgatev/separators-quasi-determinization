\subsection{Sets, relations and functions}

\begin{definition}
	A set is defined as usual. We indicate that a set $S$ contains an
	object $s$ by writing $s \in S$. Alternatively, we write $s
	\notin S$ to show that $s$ is not a member of $S$. The cardinality of
	a set $S$ is denoted by $|S|$. We use $\emptyset$ to indicate the
	unique set with no members ($|\emptyset| = 0)$.
\end{definition}

\begin{definition}
	A set $S'$ is a subset of a set $S''$ if and only if $s \in S'$ implies 
	$s \in S''$ for every $s$. It follows that $\emptyset$ is a subset of every
	set. We write $S' \subseteq S''$ to indicate that $S'$ is a subset of $S''$.
\end{definition}

\begin{definition}
	A set $S'$ is equal to a set $S''$ if and only if $S' \subseteq S''$
	and $S'' \subseteq S'$. We write $S' = S''$ to indicate that $S'$ equals
	$S''$.
\end{definition}

\begin{definition}
	The powerset of a set $S$ is the set
	\begin{align*}
		2^S = \{ S' \subseteq S \}.
	\end{align*}
\end{definition}

\begin{definition}
	The union of the sets $S_1, S_2, \ldots, S_n$ for $n \in \mathbb{N}$ is
	the set
	\begin{align*}
		\bigcup_{i=1}^n{S_i} = \{ s \mid \exists i \in \interval{1}{n}(s
		\in S_i) \}.
	\end{align*}
	We shall also use the notation $S_1 \cup S_2 \cup \ldots \cup S_n$ to
	indicate the union of sets.
\end{definition}

\begin{definition}
	The intersection of the sets $S_1, S_2, \ldots, S_n$ for $n \in \mathbb{N}$
	is the set
	\begin{align*}
		\bigcap_{i=1}^n{S_i} = \{ s \mid \forall i \in \interval{1}{n}(s
		\in S_i) \}.
	\end{align*}
	We shall also use the notation $S_1 \cap S_2 \cap \ldots \cap S_n$ to
	indicate the intersection of sets.
\end{definition}

\begin{definition}
	The sets $S_1, S_2, \ldots, S_n$ form a partition of the set $S$ for $n \in
	\mathbb{N}$ if and only if
	\begin{itemize}
		\item $S = \bigcup_{i=1}^n{S_i}$,
		\item $|S_i| > 0$ for $i \in \interval{1}{n}$,
		\item $S_i \cap S_j = \emptyset$ for $i, j \in \interval{1}{n}, i
			\neq j$.
	\end{itemize}
	We write $P^S$ to indicate that the set $P = \{ S_1, S_2, \ldots, S_n \}$
	is a partition of the set $S$.
\end{definition}

\begin{definition}
	A tuple is defined as usual. We write $\vec{t} = \langle t_1, t_2, \ldots,
	t_n \rangle$ for the tuple of the $n$ objects $t_1, t_2, \ldots, t_n$.
\end{definition}

\begin{definition}
	The Cartesian product of the sets $S_1, S_2, \ldots, S_n$ for $n \in
	\mathbb{N}$ is defined as the set
	\begin{align*}
		\prod_{i=1}^n{S_i} = \{ \langle s_1, s_2, \ldots, s_n \rangle \mid
		\forall i \in \interval{1}{n}(s_i \in S_i) \}.
	\end{align*}
	We shall also use the notation $S_1 \times S_2 \times \ldots \times S_n$ to
	indicate the Cartesian product of sets. In case $S_1 = S_2 = \ldots = S_n =
	S$ we shall write $S^n$ for the Cartesian product of the set $S$.
\end{definition}

\begin{definition}
	An $n$-ary relation $R$ on the sets $S_1, S_2, \ldots, S_n$ for $n \in
	\mathbb{N}$ is a subset
	\begin{align*}
		R \subseteq \prod_{i=1}^n{S_i}.
	\end{align*}
\end{definition}

\begin{definition}
	A binary relation $R \subseteq X \times Y$ is a function if and only if
	\begin{align*}
		\forall \langle x', y' \rangle, \langle x'', y'' \rangle \in R
		(y' \neq y'' \implies x' \neq x'').
	\end{align*}
	We write $R : X \rightarrow Y$ to indicate that the relation $R \subseteq X
	\times Y$ is a function and $R(x) = y$ in case $\langle x, y \rangle \in R$.
\end{definition}

\begin{definition}
	Let $f: X \rightarrow Y$ be a function. The domain $dom(f)$ and range
	$rng(f)$ of $f$ are the sets respectively defined as
	\begin{align*}
		dom(f) &= \{ x \in X \mid \exists y \in Y (f(x) = y) \},\\
		rng(f) &= \{ y \in Y \mid \exists x \in X (f(x) = y) \}.
	\end{align*}
\end{definition}

\begin{definition}
	A function $f: X \rightarrow Y$ is injective if and only if
	\begin{align*}
		\forall x', x'' \in dom(f) (x' \neq x'' \implies f(x') \neq f(x'')).
	\end{align*}
\end{definition}

\begin{definition}
	Let $f: X \rightarrow Y$ be a function. The set-lifted function $\hat{f}
	: 2^X \rightarrow 2^Y$ is defined as
	\begin{align*}
		\hat{f}(X') = \{ f(x) \mid x \in X' \cap dom(f) \}.
	\end{align*}
\end{definition}

\begin{definition}
	Let $f: \mathbb{N} \rightarrow \mathbb{R}$ and $g: \mathbb{N} \rightarrow
	\mathbb{R}$ be functions. We shall write $f \in \mathcal{O}(g)$ if and only
	if
	\begin{align*}
		\exists c > 0 \exists m \in \mathbb{N} \forall n > m
		(|f(n)| \leq c|g(n)|).
	\end{align*}
\end{definition}
