\subsection{Sets, relations and functions}

\begin{definition}
	A set is defined as usual. We indicate that a set $S$ contains an
	object $s$ by writing $s \in S$. Alternatively, we write $s
	\notin S$ to show that $s$ is not a member of $S$. The cardinality of
	a set $S$ is denoted by $|S|$. We use $\varnothing$ to indicate the
	unique set with no members ($|\varnothing| = 0)$.
\end{definition}

\begin{definition}
	A set $S'$ is a subset of a set $S''$ if and only if $\forall s \in S'
	(s \in S'')$. It follows that $\varnothing$ is a subset of every set. We
	write $S' \subseteq S''$ to indicate that $S'$ is a subset of $S''$.
\end{definition}

\begin{definition}
	A set $S'$ is equal to a set $S''$ if and only if $S' \subseteq S''$
	and $S'' \subseteq S'$. We write $S' = S''$ to indicate that $S'$ is equal
	to $S''$.
\end{definition}

\begin{definition}
	The powerset of a set $S$ is the set $2^S$ consisting of all subsets of $S$.
\end{definition}

\begin{definition}
	The union of the sets $S_1, S_2, \ldots, S_n$ is the set
	$\bigcup_{i=1}^n{S_i} =$\\$
	\{ s \mid \exists i \in \interval{1}{n}(s \in S_i) \}$. We shall also
	use the notation $S_1 \cup S_2 \cup \ldots \cup S_n$ to indicate the
	union of sets.
\end{definition}

\begin{definition}
	The intersection of the sets $S_1, S_2, \ldots, S_n$ is the set
	$\bigcap_{i=1}^n{S_i} = \{ s \mid \forall i \in \interval{1}{n}(s
	\in S_i) \}$. We shall also use the notation $S_1 \cap S_2 \cap \ldots
	\cap S_n$ to indicate the intersection of sets.
\end{definition}

\begin{definition}
	The sets $S_1, S_2, \ldots, S_n$ form a partition of a set $S$ if and only
	if $S = \bigcup_{i=1}^n{S_i}$, $\forall i \in \interval{1}{n}(|S_i| > 0)$
	and $\forall i, j \in \interval{1}{n} (i \neq j \implies S_i \cap S_j =
	\varnothing)$. We write $P^S$ to indicate that the set $P = \{ S_1, S_2,
	\ldots, S_n \}$ is a partition of the set $S$.
\end{definition}

\begin{definition}
	A tuple is defined as usual. We write $\langle t_1, t_2, \ldots, t_n
	\rangle$ for the tuple of the $n$ objects $t_1, t_2, \ldots, t_n$.
\end{definition}

\begin{definition}
	The Cartesian product of the sets $S_1, S_2, \ldots, S_n$ is defined
	as the set $\prod_{i=1}^n{S_i} = \{ \langle s_1, s_2, \ldots, s_n
	\rangle \mid \forall i \in \interval{1}{n}(s_i \in S_i) \}$. We shall
	also use the notation $S_1 \times S_2 \times \ldots \times S_n$ to
	indicate the Cartesian product of sets.
\end{definition}

\begin{definition}
	An $n$-ary relation $R$ on the sets $S_1, S_2, \ldots, S_n$ is a subset
	$R \subseteq \prod_{i=1}^n{S_i}$.
\end{definition}

\begin{definition}
	A binary relation $R \subseteq X \times Y$ is a function if and only if
	$\forall \langle x', y' \rangle, \langle x'', y'' \rangle \in R (y' \neq
	y'' \implies x' \neq x'')$. We write $R : X \rightarrow Y$ to indicate that
	the relation $R \subseteq X \times Y$ is a function and $R(x) = y$ in case
	$\langle x, y \rangle \in R$.
\end{definition}


\begin{definition}
	Let $f: X \rightarrow Y$ be a function. The domain of $f$ is the set
	$Dom(f) = \{ x \in X \mid \exists y \in Y (f(x) = y) \}$. The
	range of $f$ is the set $Rng(f) = \{ y \in Y \mid \exists x \in X
	(f(x) = y) \}$.
\end{definition}

\begin{definition}
	A function $f: X \rightarrow Y$ is injective if and only if $\forall x', x''
	\in Dom(f) (x' \neq x'' \implies f(x') \neq f(x''))$.
\end{definition}

\begin{definition}
	Let $f: X \rightarrow Y$ be a function. The set-lifted function $\hat{f}
	: 2^X \rightarrow 2^Y$ is defined as $\hat{f}(X') = \{ y \in Y \mid \exists
	x \in X' (f(x) = y) \}$ for $X' \subseteq X$.
\end{definition}
