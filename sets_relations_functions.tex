\section{Sets, relations, functions and tuples}

Sets, relations, functions and tuples as well as boolean operations on sets are
introduced as usual. The cardinality of a set $S$ is denoted by $|S|$. The powerset
of $S$ is denoted by $2^S$. The unique set with no members is denoted by
$\emptyset$. The sets of natural and real numbers are respectively denoted by
$\mathbb{N}$ and $\mathbb{R}$. The set of all natural numbers greater than or equal
to $m \in \mathbb{N}$ and less than or equal to $n \in \mathbb{N}$ is denoted by $\{
m, m+1, \ldots, n \}$. A tuple $t$ of $n \in \mathbb{N}$ elements is denoted by
$\langle t_1, t_2, \ldots, t_n \rangle$ and its length by $|t|$.

\begin{definition}
	The sets $S_1, S_2, \ldots, S_n$ for $n > 0$ form a \emph{partition of a
	set} $S$ if and only if
	\begin{itemize}
	\item $S_i \neq \emptyset$ for $i \in \{ 1, 2, \ldots, n \}$,
	\item $S_i \cap S_j = \emptyset$ if $i \neq j$ for $i, j \in \{ 1, 2,
		\ldots, n \}$,
	\item $S = \bigcup_{i=1}^n S_i$.
	\end{itemize}
\end{definition}

We shall write ${\{ S_1, S_2, \ldots, S_n \}}_S$ to indicate that the sets $S_1, S_2,
\ldots, S_n$ form a partition of the set $S$.

\begin{definition}
	The \emph{domain and range of a function} $f: X \rightarrow Y$ are the sets
	respectively defined as
	\begin{align*}
	dom(f) &= \{ x \in X \mid \exists y \in Y (f(x) = y) \},\\
	rng(f) &= \{ y \in Y \mid \exists x \in X (f(x) = y) \}.
	\end{align*}
\end{definition}

\begin{definition}
	A function $f: X \rightarrow Y$ is \emph{injective} if and only if $x' \neq
	x''$ implies $f(x') \neq f(x'')$ for $x', x'' \in dom(f)$.
\end{definition}

\begin{definition}
	The \emph{set-lifted function} $\hat{f} : 2^X \rightarrow 2^Y$ induced by a
	function $f: X \rightarrow Y$ is defined as
	\begin{equation*}
	\hat{f}(Z) = \{ f(x) \in Y \mid \exists x \in Z \cap dom(f) \}.
	\end{equation*}
\end{definition}

\begin{definition}
	Let $g : \mathbb{N} \rightarrow \mathbb{R}$ be a function. Then
	$\mathcal{O}(g)$ denotes the set of functions $f: \mathbb{N} \rightarrow
	\mathbb{R}$ such that $|f(n)| \leq c|g(n)|$ for $n \geq m$ where $c > 0$
	and $m \in \mathbb{N}$.
\end{definition}
