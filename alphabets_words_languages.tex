\subsection{Alphabets, words and languages}

\begin{definition}
	An alphabet $\Sigma$ is a finite set.
\end{definition}

\begin{definition}
	A word $w$ of length $n \in \mathbb{N}$ over an alphabet $\Sigma$ is
	a tuple $\langle \sigma_1, \sigma_2, \ldots, \sigma_n \rangle$ where
	$\sigma_i \in \Sigma$ for $i \in \{ 1, 2, \ldots, n \}$. The notation
	$\sigma_1\sigma_2\ldots\sigma_n$ is preferred over $\langle \sigma_1,
	\sigma_2, \ldots, \sigma_n \rangle$ for words.
\end{definition}

\begin{definition}
	The unique word of length $0$ over any alphabet is denoted by $\epsilon$.
\end{definition}

\begin{definition}
	Let $\Sigma$ be an alphabet. The set $\Sigma \cup \{ \epsilon \}$ is denoted
	by $\Sigma_\epsilon$.
\end{definition}

\begin{definition}
	Let $\Sigma$ be an alphabet. The set of all words of length $n$ over $\Sigma$
	is denoted by $\Sigma^n = \Sigma^{n-1} \times \Sigma$ where $\Sigma^0 = \{
	\epsilon \}$. The set of all words over $\Sigma$ is denoted by $\Sigma^* =
	\bigcup_{i = 0}^{\infty} \Sigma^i$.
\end{definition}

\begin{definition}
	Concatenation of words over an alphabet $\Sigma$ is the function $\cdot :
	\Sigma^* \times \Sigma^* \rightarrow \Sigma^*$ defined as
	\begin{align*}
		w' \cdot w'' = \sigma_1' \ldots \sigma_n' \sigma_1'' \ldots
		\sigma_m''
	\end{align*}
	where $w' = \sigma'_1\sigma'_2\ldots\sigma'_n$ and $w'' =
	\sigma''_1\sigma''_2\ldots\sigma''_m$ are words over $\Sigma$. It follows
	that $w \cdot \epsilon = \epsilon \cdot w = w$ for any word $w$.
	Concatenation of words is associative, that is $u \cdot (v \cdot w) = (u
	\cdot v) \cdot w$ for all words $u$, $v$ and $w$. Thus, we shall from now on
	write $w'w''$ instead of $w' \cdot w''$ without causing ambiguity.
\end{definition}

\begin{definition}
	A language $\mathcal{L}$ over an alphabet $\Sigma$ is a subset
	\begin{align*}
		\mathcal{L} \subseteq \Sigma^*.
	\end{align*}
\end{definition}

\begin{definition}
	Let $\mathcal{L}$ be a language over an alphabet $\Sigma$. The $n$-iterated
	language induced by $\mathcal{L}$ is the set $\mathcal{L}^n =
	\mathcal{L}^{n-1} \hat{\cdot} \mathcal{L}$ where $\mathcal{L}^0 = \{
	\epsilon \}$. The iterated language induced by $\mathcal{L}$ is the set
	$\mathcal{L}^* = \bigcup_{i = 0}^{\infty} \mathcal{L}^i$.
\end{definition}
