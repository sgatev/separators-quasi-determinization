\subsection{Alphabets, words and languages}

\begin{definition}
	An alphabet $\Sigma$ is a finite set of characters.
\end{definition}

\begin{definition}
	A word $w$ of length $n \in \mathbb{N}$ over an alphabet $\Sigma$ is
	a tuple $\langle \sigma_1, \sigma_2, \ldots, \sigma_n \rangle$ where
	$\sigma_i \in \Sigma$ for $i \in \interval{1}{n}$. The notation
	$\sigma_1\sigma_2\ldots\sigma_n$ is preferred over $\langle \sigma_1,
	\sigma_2, \ldots, \sigma_n \rangle$ for words. The length of the word
	$w$ is denoted by $|w|$.
\end{definition}

\begin{definition}
	We use $\epsilon$ to denote the unique word of length $0$ over any alphabet.
\end{definition}

\begin{definition}
	Let $\Sigma$ be an alphabet. We use $\Sigma_\epsilon$ to denote the set
	$\Sigma \cup \{ \epsilon \}$.
\end{definition}

\begin{definition}
	Let $\Sigma$ be an alphabet. We use $\Sigma^n = \Sigma^{n-1} \times \Sigma$
	for $n \in \mathbb{N}$ to denote the set of all words of length $n$ over
	$\Sigma$ where $\Sigma^0 = \{ \epsilon \}$. We write $\Sigma^* = \bigcup_{i
	= 0}^{\infty} \Sigma^i$ for the set of all words over $\Sigma$.
\end{definition}

\begin{definition}
	Concatenation of words over an alphabet $\Sigma$ is the function $\cdot :
	\Sigma^* \times \Sigma^* \rightarrow \Sigma^*$ defined as
	\begin{align*}
		w' \cdot w'' = \sigma_1' \ldots \sigma_n' \sigma_1'' \ldots
		\sigma_m''
	\end{align*}
	where $w' = \sigma'_1\sigma'_2\ldots\sigma'_n$ is a word of length $n$ over
	$\Sigma$ and $w'' = \sigma''_1\sigma''_2\ldots\sigma''_m$ is a word of
	length $m$ over $\Sigma$. The concatenation $w' \cdot w''$ is a word of
	length $n+m$ over $\Sigma$. It follows that $w \cdot \epsilon = \epsilon
	\cdot w = w$ for any word $w$. Concatenation of words is an associative
	operation. That is, $u \cdot (v \cdot w) = (u \cdot v) \cdot w$ for all
	words $u$, $v$ and $w$. Thus, we shall from now on write $w'w''$ instead of
	$w' \cdot w''$ without causing ambiguity.
\end{definition}

\begin{definition}
	Let $w$ be a word over an alphabet $\Sigma$. The word $u$ is a prefix of
	$w$ if and only if $w = uv$ for some $v \in \Sigma^*$. It follows that
	$\epsilon$ is a prefix of every word. The set of all prefixes of the word
	$w$ is denoted by
	\begin{align*}
		Pref(w) = \{ u \in \Sigma^* \mid \exists v \in \Sigma^* (w = uv) \}.
	\end{align*}
\end{definition}

\begin{definition}
	Let $w = uv$ be a word over an alphabet $\Sigma$. We use $u^{-1}w$ to denote
	the word $v$ derived by removing the prefix $u$ from $w$. It follows that
	$\epsilon^{-1}w = w$ for any word $w$.
\end{definition}

\begin{proposition}
	Let $u$ and $v$ be words over an alphabet $\Sigma$. Then
	\begin{align*}
		v^{-1}u^{-1}w = (uv)^{-1}w
	\end{align*}
	for every $w \in \Sigma^*$.
\end{proposition}

\begin{proof}
	We prove the statement by induction on $v$. Let $v = \epsilon$. Then
	\begin{align*}
		v^{-1}u^{-1}w
		&= \epsilon^{-1}u^{-1}w\\
		&= u^{-1}w\\
		&= (u\epsilon)^{-1}w\\
		&= (uv)^{-1}w.
	\end{align*}
	Let us assume that the statement holds for $v' \in \Sigma^*$. Let $v = v'a$
	where $a \in \Sigma$. Then
	\begin{align*}
		v^{-1}u^{-1}w
		&= (v'a)^{-1}u^{-1}w\\
		&= a^{-1}v'^{-1}u^{-1}w\\
		&= a^{-1}(uv')^{-1}w\\
		&= (uv'a)^{-1}w\\
		&= (uv)^{-1}w.
	\end{align*}
\end{proof}

\begin{definition}
	The longest common prefix of the words $w_1, w_2, \ldots, w_n$ for $n \in
	\mathbb{N}$ is the word
	\begin{align*}
		\bigwedge_{i=1}^n w_i = \argmax_{u \in \bigcap_{i=1}^n{Pref(w_i)}}{
		|u|}.
	\end{align*}
	We shall also use the notation $w_1 \wedge w_2 \wedge \ldots \wedge w_n$ to
	indicate the longest common prefix of words.
\end{definition}

\begin{definition}
	Sequential distance of words over an alphabet $\Sigma$ is the function
	$d_\Sigma : \Sigma^* \times \Sigma^* \rightarrow \mathbb{N}$ defined as
	\begin{align*}
		d_\Sigma(u, v) = |u| + |v| - 2|u \wedge v|.
	\end{align*}
\end{definition}

\begin{definition}
	Let $\Sigma$ and $\Omega$ be alphabets. A function $f : \Sigma^*
	\rightarrow \Omega^*$ has bounded variation if and only if
	\begin{align*}
		\forall k \geq 0 \exists K \geq 0 \forall u, v \in dom(f)
		(d_S(u, v) \leq k \implies d_S(f(u), f(v)) \leq K).
	\end{align*}
\end{definition}

\begin{definition}
	A language $L$ over an alphabet $\Sigma$ is a subset
	\begin{align*}
		L \subseteq \Sigma^*.
	\end{align*}
\end{definition}

\begin{definition}
	The class of regular languages over an alphabet $\Sigma$ is defined by
	induction:
	\begin{enumerate}
		\item $\emptyset$ is a regular language over $\Sigma$;
		\item if $\sigma \in \Sigma_\epsilon$ then $\{ \sigma \}$ is a
			regular language over $\Sigma$;
		\item if $L_1$ and $L_2$ are regular languages over $\Sigma$ then
			\begin{itemize}
				\item $L_1 \hat{\cdot} L_2$ is a regular language
					over $\Sigma$,
				\item $L_1 \cup L_2$ is a regular language over
					$\Sigma$,
				\item $L_1^*$ is a regular language over $\Sigma$;
			\end{itemize}
		\item there are no other regular languages over $\Sigma$.
	\end{enumerate}
\end{definition}
