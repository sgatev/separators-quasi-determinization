\subsection{Alphabets, words and languages}

\begin{definition}
	An alphabet $\Sigma$ is a finite set of characters.
\end{definition}

\begin{definition}
	A word $w$ of length $n \in \mathbb{N}$ over an alphabet $\Sigma$ is
	a tuple $\langle \sigma_1, \sigma_2, \ldots, \sigma_n \rangle$ where
	$\forall i \in \interval{1}{n}(\sigma_i \in \Sigma)$. The notation
	$\sigma_1\sigma_2\ldots\sigma_n$ is preferred over $\langle \sigma_1,
	\sigma_2, \ldots, \sigma_n \rangle$ for words. The length of the word
	$w$ is denoted by $|w|$.
\end{definition}

\begin{definition}
	We use $\epsilon$ to denote the unique word of length $0$ over any alphabet.
\end{definition}

\begin{definition}
	Let $\Sigma$ be an alphabet. We use $\Sigma_\epsilon$ to denote the set
	$\Sigma \cup \{ \epsilon \}$.
\end{definition}

\begin{definition}
	Let $\Sigma$ be an alphabet. We use $\Sigma^n = \Sigma^{n-1} \times \Sigma$
	to denote the set of all words of length $n$ over $\Sigma$ where $\Sigma^0
	= \{ \epsilon \}$. We use $\Sigma^* = \bigcup_{i = 0}^{\infty} \Sigma^i$ to
	denote the set of all words over $\Sigma$.
\end{definition}

\begin{definition}
	Concatenation of words over an alphabet $\Sigma$ is the function $\cdot :
	\Sigma^* \times \Sigma^* \rightarrow \Sigma^*$ defined as
	\begin{align*}
		w' \cdot w'' = \sigma_1' \ldots \sigma_n' \sigma_1'' \ldots
		\sigma_m''
	\end{align*}
	where $w' = \sigma'_1\sigma'_2\ldots\sigma'_n$ is a word of length $n$ over
	$\Sigma$ and $w'' = \sigma''_1\sigma''_2\ldots\sigma''_m$ is a word of
	length $m$ over $\Sigma$. The concatenation $w' \cdot w''$ is a word of
	length $n+m$ over $\Sigma$. It follows that $w \cdot \epsilon = \epsilon
	\cdot w = w$ for any word $w$. Concatenation of words is an associative
	operation. That is, $u \cdot (v \cdot w) = (u \cdot v) \cdot w$ for all
	words $u$, $v$ and $w$. Thus, we shall from now on write $w'w''$ instead of
	$w' \cdot w''$.
\end{definition}

\begin{definition}
	Let $w$ be a word over an alphabet $\Sigma$. The word $u$ is prefix of
	$w$ if and only if $\exists v \in \Sigma^* (w = uv)$. It follows that
	$\epsilon$ is a prefix of every word. We use $Pref(w) = \{ u \in \Sigma^*
	\mid \exists v \in \Sigma^* (w = uv) \}$ to denote the set of all prefixes
	of the word $w$.
\end{definition}

\begin{definition}
	Let $w = uv$ be a word over an alphabet $\Sigma$. We use $u^{-1}w$ to denote
	the word $v$ derived by removing the prefix $u$ from $w$. It follows that
	$\epsilon^{-1}w = w$ for any word $w$.
\end{definition}

\begin{definition}
	The longest common prefix of the words $w_1, w_2, \ldots, w_n$ is the word
	\begin{align*}
		\bigwedge_{i=1}^n w_i = \argmax_{u \in \bigcap_{i=1}^n{Pref(w_i)}}{
		|u|}.
	\end{align*}
	We shall also use the notation $w_1 \wedge w_2 \wedge \ldots \wedge w_n$ to
	indicate the longest common prefix of words.
\end{definition}

\begin{definition}
	A language $L$ over an alphabet $\Sigma$ is a subset $L \subseteq \Sigma^*$.
\end{definition}

\begin{definition}
	A language $L$ over an alphabet $\Sigma$ is regular if and only if it can be
	constructed inductively as follows
	\begin{enumerate}
		\item $\emptyset$ is a regular language over $\Sigma$;
		\item If $\sigma \in \Sigma_\epsilon$ then $\{ \sigma \}$
			is a regular language over $\Sigma$;
		\item If $L'$ and $L''$ are regular languages over $\Sigma$ then
			\begin{itemize}
				\item $L' \hat{\cdot} L''$ is a regular language
					over $\Sigma$;
				\item $L' \cup L''$ is a regular language over
					$\Sigma$;
				\item $L'^*$ is a regular language over $\Sigma$;
			\end{itemize}
		\item There are no other regular languages over $\Sigma$.
	\end{enumerate}
\end{definition}
