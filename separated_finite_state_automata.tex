\subsection{Separated finite-state automata}

\begin{definition}
	A finite-state automaton $\mathcal{A} = \langle \Sigma, Q, S, F, \Delta
	\rangle$ is $k$-separated for $k > 1$ if and only if there exists a state
	separator subset $X \subseteq Q$ inducing a partition $P_Q = \{ X, Q_1, Q_2,
	\ldots, Q_k \}$ such that $\Delta \cap (Q_i \times \Sigma_\epsilon \times
	Q_j) = \emptyset$ if $i \neq j$ for $i, j \in \interval{1}{k}$. The
	$k$-separated finite-state automaton $\mathcal{A}$ is denoted by
	\begin{align*}
		\mathcal{A}_k = \langle \Sigma, \vec{Q}, X, S, F, \Delta \rangle.
	\end{align*}
\end{definition}

\begin{proposition}
	Let $\mathcal{A} = \langle \Sigma, Q, S, F, \Delta \rangle$ be a
	finite-state automaton and let $X$ be a separator in
	$\mathcal{G}_\mathcal{A}$ inducing a partition $P_Q = \{ X, Q_1, Q_2, \ldots,
	Q_k \}$. Then $\mathcal{A}$ is $k$-separated and $\mathcal{A}_k = \langle
	\Sigma, \vec{Q}, X, S, F, \Delta \rangle$.
\end{proposition}

\begin{proof}
	Since $X$ is a separator in $\mathcal{G}_\mathcal{A}$ then $E \cap (Q_i
	\times Q_j) = \emptyset$ if $i \neq j$ for $i, j \in \interval{1}{k}$ where
	$E = \{ \langle q', q'' \rangle \in Q \times Q \mid \exists \langle q',
	\sigma, q'' \rangle \in \Delta \}$. Thus, $\Delta \cap (Q_i \times
	\Sigma_\epsilon \times Q_j) = \emptyset$ if $i \neq j$ for $i, j \in
	\interval{1}{k}$. Therefore, $\mathcal{A}$ is $k$-separated and
	$\mathcal{A}_k = \langle \Sigma, \vec{Q}, X, S, F, \Delta \rangle$.
\end{proof}

\begin{proposition}\label{p_small_delta}
	Let $\mathcal{A}_k = \langle \Sigma, \vec{Q}, X, S, F, \Delta \rangle$ be a
	$k$-separated finite-state automaton. Then $\Delta \subseteq \bigcup_{i=1}^k
	(Q_i \cup X) \times \Sigma_\epsilon \times (Q_i \cup X)$.
\end{proposition}

\begin{proof}
	By definition we have
	\begin{flalign*}
		\Delta &\subseteq Q \times \Sigma_\epsilon \times Q &\\
		&= (\bigcup_{i=1}^k{Q_i} \cup X) \times \Sigma_\epsilon \times
		(\bigcup_{i=1}^k{Q_i} \cup X)\\
		&= \begin{aligned}[t]
			&\bigcup_{i=1}^k \bigcup_{j=1}^k (Q_i \times
			\Sigma_\epsilon \times Q_j) \cup\\
			&\bigcup_{i=1}^k (X \times \Sigma_\epsilon \times Q_i) \cup
			\bigcup_{i=1}^k (Q_i \times \Sigma_\epsilon \times X) \cup\\
			&(X \times \Sigma_\epsilon \times X).
		\end{aligned}
	\end{flalign*}
	However, since $\mathcal{A}$ is $k$-separated $\Delta \cap (Q_i \times
	\Sigma_\epsilon \times Q_j) = \emptyset$ if $i \neq j$ for $i, j \in
	\interval{1}{k}$. Therefore,
	\begin{flalign*}
		\Delta &\subseteq \begin{aligned}[t]
			&\bigcup_{i=1}^k (Q_i \times \Sigma_\epsilon \times Q_i)
			\cup\\
			&\bigcup_{i=1}^k (X \times \Sigma_\epsilon \times Q_i) \cup
			\bigcup_{i=1}^k (Q_i \times \Sigma_\epsilon \times X) \cup\\
			&(X \times \Sigma_\epsilon \times X)
		\end{aligned} &\\
		&= \bigcup_{i=1}^k ((Q_i \cup X) \times \Sigma_\epsilon \times Q_i)
		\cup \bigcup_{i=1}^k ((Q_i \cup X) \times \Sigma_\epsilon \times X)\\
		&= \bigcup_{i=1}^k (Q_i \cup X) \times \Sigma_\epsilon \times
		(Q_i \cup X).
	\end{flalign*}
\end{proof}

We know that the states of a deterministic finite-state automaton $\mathcal{A'} =
\langle \Sigma, Q', s, F', \delta \rangle$ derived by applying the construction from
theorem \ref{t_rabin_scott} are subsets of the states of the source finite-state
automaton $\mathcal{A} = \langle \Sigma, Q, S, F, \Delta \rangle$. Thus, $|\delta|
\leq |Q'| \times |\Sigma| = 2^{|Q|} \times |\Sigma|$. However, based on proposition
\ref{p_small_delta}, if $\mathcal{A}$ is $k$-separated for $k > 1$ we could
effectively simulate $p = \delta(q, \sigma)$ by decomposing the state transition
function $\delta$ into functions $\delta_i : (Q_i \cup X) \times \Sigma \rightarrow
(Q_i \cup X)$ for $i \in \interval{1}{k}$ and the state $q \subseteq Q$ into
quasi-states $q_i \subseteq Q_i \cup X$ for $i \in \interval{1}{k}$, evaluating $p_i
= \delta_i(q_i, \sigma)$ for each of the quasi-states and composing the results into
the new state $p = \bigcup_{i=1}^k q_i \subseteq Q$. If $\sum_{i=1}^k 2^{|Q_i| +
|X|} < 2^{|Q|}$ the functions $\delta_i$ will be smaller than $\delta$ leading
towards a more efficient representation of $\mathcal{A'}$. This insight motivates
the constructions described below.
