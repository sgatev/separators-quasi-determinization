\subsection{Separated finite-state automata}

\begin{definition}
	A finite-state automaton $\mathcal{A} = \langle \Sigma, Q, S, F, \Delta
	\rangle$ is $k$-separated for $k \in \mathbb{N}$ if and only if there
	exists a separator $X$ in the graph $\mathcal{G}_\mathcal{A} = \langle Q,
	\Sigma_\epsilon, \Delta \rangle$ inducing a partition $Q_X = \{ Q_1, Q_2,
	\ldots, Q_k \}$. We write $\mathcal{A} = \langle \Sigma, \vec{Q}, X, S, F,
	\Delta \rangle$ to denote the $k$-separated finite-state automaton
	$\mathcal{A}$.
\end{definition}

\begin{proposition}
	Let $\mathcal{A} = \langle \Sigma, \vec{Q}, X, S, F, \Delta \rangle$ be a
	$k$-separated finite-state automaton. Then $\Delta \subseteq \bigcup_{i=1}^k
	(Q_i \cup X) \times \Sigma_\epsilon \times (Q_i \cup X)$.
\end{proposition}

\begin{proof}
	\begin{align*}
		\Delta &\subseteq Q \times \Sigma_\epsilon \times Q\\
		&= (\bigcup_{i=1}^k{Q_i} \cup X) \times \Sigma_\epsilon \times
		(\bigcup_{i=1}^k{Q_i} \cup X)\\
		&= \begin{aligned}[t]
			&\bigcup_{i=1}^k \bigcup_{j=1}^k (Q_i \times
			\Sigma_\epsilon \times Q_j) \cup\\
			&\bigcup_{i=1}^k (X \times \Sigma_\epsilon \times Q_i) \cup
			(Q_i \times \Sigma_\epsilon \times X) \cup\\
			&(X \times \Sigma_\epsilon \times X)
		\end{aligned}
	\end{align*}
	However, $Q_i \times \Sigma_\epsilon \times Q_j = \emptyset$ for $i \neq j$.
	Therefore,
	\begin{align*}
		\Delta &\subseteq \begin{aligned}[t]
			\bigcup_{i=1}^k &(Q_i \times \Sigma_\epsilon \times Q_i)
			\cup (X \times \Sigma_\epsilon \times Q_i) \cup\\
			&(Q_i \times \Sigma_\epsilon \times X) \cup (X \times
			\Sigma_\epsilon \times X)
		\end{aligned}\\
		&= \bigcup_{i=1}^k \big[ (Q_i \cup X) \times \Sigma_\epsilon
		\times Q_i \big] \cup \big[ (Q_i \cup X) \times \Sigma_\epsilon
		\times X \big]\\
		&= \bigcup_{i=1}^k (Q_i \cup X) \times \Sigma_\epsilon \times
		(Q_i \cup X)
	\end{align*}
\end{proof}

\begin{proposition}
	Let $\mathcal{A} = \langle \Sigma, \vec{Q}, X, S, F, \Delta \rangle$ be a
	$k$-separated finite-state automaton. Let $\Delta_i = \Delta \cap \big[ (Q_i
	\cup X) \times \Sigma_\epsilon \times (Q_i \cup X) \big]$ for $i \in
	\interval{1}{k}$. Then $\Delta(q, \sigma) = \Delta_i(q, \sigma)$ for $i \in
	\interval{1}{k}$, $q \in Q_i \cup X$ and $\sigma \in \Sigma_\epsilon$.
\end{proposition}

\begin{proof}
	We have $\langle q, \sigma, p \rangle \in \Delta \implies \langle q, \sigma,
	p \rangle \in \Delta \cap \big[ \bigcup_{i=1}^k (Q_i \cup X) \times
	\Sigma_\epsilon \times (Q_i \cup X) \big] \implies \langle q, \sigma, p
	\rangle \in \Delta \cap \big[ (Q_i \cup X) \times \Sigma_\epsilon \times
	(Q_i \cup X) \big] \implies \langle q, \sigma, p \rangle \in \Delta_i$.

	Similarly, $\langle q, \sigma, p \rangle \in \Delta_i \implies \langle q,
	\sigma, p \rangle \in \bigcup_{i=1}^k \Delta_i \implies \langle q, \sigma,
	p \rangle \in \Delta$.
\end{proof}

\begin{proposition}
	Let $\mathcal{A} = \langle \Sigma, \vec{Q}, X, S, F, \Delta \rangle$ be a
	$k$-separated finite-state automaton. Let $\Delta_i = \Delta \cap \big[ (Q_i
	\cup X) \times \Sigma_\epsilon \times (Q_i \cup X) \big]$ for $i \in
	\interval{1}{k}$. Then for $P \subseteq Q$ and $\sigma \in \Sigma_\epsilon$
	we have $\hat{\Delta}(P, \{ \sigma \}) = \bigcup_{i=1}^k \hat{\Delta}_i(P_i,
	\{ \sigma \})$ where $P_i = P \cap (Q_i \cup X)$ for $i \in \interval{1}{k}$.
\end{proposition}

\begin{proof}
	\begin{align*}
		\hat{\Delta}(P, \{ \sigma \})
		&= \bigcup_{p \in P} \Delta(p, \sigma)\\
		&= \bigcup_{i=1}^k \bigcup_{p \in P_i} \Delta(p, \sigma)\\
		&= \bigcup_{i=1}^k \bigcup_{p \in P_i} \Delta_i(p, \sigma)
		= \bigcup_{i=1}^k \hat{\Delta}_i(P_i, \{ \sigma \})
	\end{align*}
\end{proof}
