\section{Graphs and separators}

\begin{definition}
  A \emph{graph} $\mathcal{G}$ is a pair
  \begin{equation*}
    \langle V, E \rangle
  \end{equation*}
  where
  \begin{itemize}
    \item $V$ is a finite set of vertices,
    \item $E$ is a finite set of edges,
    \item $e \subseteq V$ and $|e| \in \{ 1, 2 \}$ for $e \in E$.
  \end{itemize}
\end{definition}

\begin{definition}
  The set $X$ is a \emph{separator in a graph} $\mathcal{G} = \langle V, E \rangle$
  if and only if there exists a partition ${\{ X, V_1, \ldots, V_n \}}_V$ for $n > 1$
  such that $\{ v', v'' \} \not\in E$ for $v' \in V_i$ and $v'' \in V_j$ if $i
  \neq j$ for $i, j \in \{1, 2, \ldots, n \}$.
\end{definition}

A \emph{two-terminal graph} is a graph $\mathcal{G} = \langle V, E \rangle$ together with
two distinct vertices $a, b \in V$, its terminals, written $\langle \mathcal{G}, a, b
\rangle$. Let $\langle \langle V_1, E_1 \rangle, a_1, b_1 \rangle$ and $\langle \langle V_2,
E_2 \rangle, a_2, b_2 \rangle$ be two-terminal graphs. If $V_1 \cap V_2 = \{ b_1 \}$ and
$b_1 = a_2$, their \emph{series composition} is $\langle \langle V_1 \cup V_2, E_1 \cup E_2
\rangle, a_1, b_2 \rangle$. If $V_1 \cap V_2 = \{ a_1, b_1 \}$, $a_1 = a_2$ and $b_1 =
b_2$, their \emph{parallel composition} is $\langle \langle V_1 \cup V_2, E_1 \cup E_2
\rangle, a_1, b_1 \rangle$.

\begin{definition}\label{def:series_parallel}
  The \emph{two-terminal series--parallel graphs} are the smallest class of two-terminal
  graphs that contains $\langle \langle \{ a, b \}, \{ \{ a, b \} \} \rangle, a, b
  \rangle$ for all distinct vertices $a$ and $b$ and is closed under series and parallel
  composition.
\end{definition}

Series--parallel graphs are planar and have treewidth at most $2$
\cite{bodlaender_arboretum}, so they have balanced separators of constant size. For
finite-state automata constructed from regular expressions, separators of size $2$ are
computed in \Cref{section:regular_expressions_automata}.
