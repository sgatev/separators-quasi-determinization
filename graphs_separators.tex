\subsection{Graphs and separators}

\begin{definition}
	A graph $\mathcal{G}$ is a pair
	\begin{align*}
		\langle V, E \rangle
	\end{align*}
	where
	\begin{itemize}
		\item $V$ is a finite set of vertices,
		\item $E$ is a finite set of edges,
		\item $e \subseteq V$ and $|e| = 2$ for $e \in E$.
	\end{itemize}
\end{definition}

\begin{definition}
	A graph $\mathcal{G} = \langle V, E \rangle$ is planar if and only if
	there exists an injective function $f_v$ that maps vertices of the graph to
	distinct points in the real plane and an injective function $f_e$ that maps
	edges of the graph to simple curves in the real plane such that curves in
	$rng(f_e)$ intersect only at points in $rng(f_v)$.
\end{definition}

\begin{definition}
	Let $\mathcal{G} = \langle V, E \rangle$ be a graph. The set $X \subseteq
	V$ is a separator in $\mathcal{G}$ if and only if there exists a partition
	$P_V = \{ X, V_1, \ldots, V_n \}$ for $n > 1$ such that $E \cap \{ \{ v_i,
	v_j \} \mid \exists v_i \in V_i \; \exists v_j \in V_j \} = \emptyset$ if $i
	\neq j$ for $i, j \in \interval{1}{n}$.
\end{definition}

\begin{theorem}[Lipton, Tarjan, see \cite{LT77}]\label{t_lipton_tarjan}
	Let $\mathcal{G} = \langle V, E \rangle$ be a planar graph. There exist a
	separator $X$ and a partition $P_V = \{ X, V_1, V_2 \}$ in $\mathcal{G}$
	such that $|X| \leq 2 \sqrt{2} \sqrt{|V|}$ and $|V_i| \leq \frac{2}{3} |V|$
	for $i \in \interval{1}{2}$. Furthermore, there exists an algorithm that
	finds $P_V$ in $\mathcal{O}(|V|)$ time.
\end{theorem}
