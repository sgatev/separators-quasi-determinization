\subsection{Graphs and separators}

\begin{definition}
	A graph $\mathcal{G}$ is a triple $\langle V, L, E \rangle$ where $V$ is a
	finite set of vertices, $L$ is a finite set of edge labels and $E \subseteq
	V \times L \times V$ is a relation of edges.
\end{definition}

\begin{definition}
	Let $\mathcal{G} = \langle V, L, E \rangle$ be a graph. The set $X$ is a
	separator in $\mathcal{G}$ if and only if there exists a partition $P^V =
	\{ X, V_1, V_2, \ldots, V_n \}$ such that $\forall i, j \in \interval{1}{n}
	(i \neq j \implies E \cap (V_i \times L \times V_j) = \emptyset)$. We
	write $V_X = \{ V_1, V_2, \ldots, V_n \}$ for the partition of $V$ induced
	by $X$.
\end{definition}

\begin{definition}
	A graph $\mathcal{G} = \langle V, L, E \rangle$ is planar if and only if
	there exists an injective function $f_v : V \rightarrow \mathbb{R}^2$ that
	maps vertices of the graph to distinct points in the real plane and an
	injective function $f_e$ that maps edges of the graph to simple curves in
	the real plane such that curves intersect only at points in $Rng(f_v)$.
\end{definition}

\begin{theorem}[Lipton, Tarjan \cite{LT79}]
	Let $\mathcal{G} = \langle V, L, E \rangle$ be a planar graph. We can
	efficiently find a separator $X$ in $\mathcal{G}$ such that $V_X = \{ V_1,
	V_2 \}$, $|V_1| \leq \frac{2}{3} \times |V|$, $|V_2| \leq \frac{2}{3}
	\times |V|$ and $|X| \leq 2 \sqrt{2} \sqrt{|V|}$.
\end{theorem}
